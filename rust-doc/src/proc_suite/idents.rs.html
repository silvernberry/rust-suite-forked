16//! Deterministic identifier generation for proc-macros.
17//!
18//! Generates internal identifiers from stable textual anchors, producing
19//! repeatable identifier names without requiring runtime state or caching.
20//!
21//! Identifiers are derived from:
22//! - a type name: for type-based anchors
23//! - an identifier name: for name-based coordination
24//! - a token stream: for syntax-based coordination
25//!
26//! Each anchor is transformed directly into an identifier, so the same
27//! anchor produces the same identifier without relying on a global map.
28//!
29//! ## Usage
30//!
31//! [`MarkerIdent`] is blanket-implemented for all types:
32//!
33//! ```ignore
34//! struct MyFunction;
35//!
36//! let fn_ident = <MyFunction as MarkerIdent<8>>::from_type(
37//!     Case::Lower,
38//!     Some(b"prefix"),
39//!     None,
40//!     None,
41//! );
42//! ```
43//!
44//! ## Notes
45//!
46//! - Identifier generation is deterministic from the selected anchor
47//! - Casing via [`Case`]
48//! - Optional ASCII prefix/suffix
49//! - Optional span for hygiene
50

66/// Specifies how a generated identifier should be cased.
67///
68/// This is used when formatting internally generated identifiers so they
69/// blend naturally into Rust syntax and conventions at the expansion site.
70#[derive(Clone, Copy)]
71pub enum Case {
72    /// Lowercase, underscore-separated (`snake_case`).
73    ///
74    /// Commonly used for functions, variables, and other value-level items.
75    Lower,
76
77    /// Uppercase, underscore-separated (`SCREAMING_SNAKE_CASE`).
78    ///
79    /// Commonly used for constants and other all-caps items.
80    Upper,
81
82    /// PascalCase (leading uppercase, no separators).
83    ///
84    /// Commonly used for types and other type-level items.
85    Pascal,
86}

92/// A unified interface for generating deterministic identifiers from
93/// different identity anchors.
94///
95/// **Why this exists:**  
96/// Proc-macro code often needs to generate internal identifiers that are
97/// stable, collision-resistant, and reproducible. This trait groups the
98/// supported anchoring strategies behind a single, explicit API so call
99/// sites can derive identifiers without managing the underlying
100/// representation of the anchor.
101///
102/// **Three ways to anchor identity:**
103///
104/// - **`from_type`**  
105///   Uses the implementing type `Self` as the identity anchor. The type's
106///   [`std::any::type_name`] is used as the deterministic textual key.
107///   No value is ever constructed.
108///
109/// - **`from_ident`**  
110///   Uses an existing [`Ident`] as the identity anchor. The identifier's
111///   textual name is used as the key; its span and hygiene do not
112///   participate in identifier generation.
113///
114/// - **`from_tokens`**  
115///   Uses a [`proc_macro2::TokenStream`] as the identity anchor. Its
116///   deterministic textual representation is used as the key.
117///
118/// In all cases, the anchor is converted into a deterministic base word
119/// before the requested casing, prefix, and suffix are applied.
120///
121/// An explicit span may be provided to control the hygiene of the
122/// resulting identifier. When no span is provided, the generated
123/// identifier uses the default span produced by `format_ident!`.
124pub trait MarkerIdent: 'static + Sized {
125    /// Generates an identifier anchored to the implementing marker type.
126    fn from_type(
127        case: Case,
128        prefix: Option<&[u8]>,
129        suffix: Option<&[u8]>,
130        span: Option<Span>,
131    ) -> Ident {
132        let string = gen_ident_from_marker::<Self>(case, prefix, suffix);
133
134        if let Some(span) = span {
135            return Ident::new(&string, span);
136        }
137
138        format_ident!("{}", string)
139    }
140
141    /// Generates an identifier anchored to an existing identifier name.
142    fn from_ident(
143        ident: &Ident,
144        case: Case,
145        prefix: Option<&[u8]>,
146        suffix: Option<&[u8]>,
147        span: Option<Span>,
148    ) -> Ident {
149        let string = gen_ident_from_ident(ident, case, prefix, suffix);
150
151        if let Some(span) = span {
152            return Ident::new(&string, span);
153        }
154
155        format_ident!("{}", string)
156    }
157
158    /// Generates an identifier anchored to a token stream.
159    fn from_tokens(
160        tokens: &proc_macro2::TokenStream,
161        case: Case,
162        prefix: Option<&[u8]>,
163        suffix: Option<&[u8]>,
164        span: Option<Span>,
165    ) -> Ident {
166        let string = gen_ident_from_tokens(tokens, case, prefix, suffix);
167
168        if let Some(span) = span {
169            return Ident::new(&string, span);
170        }
171
172        format_ident!("{}", string)
173    }
174}

183/// Returns the bare (unqualified) ASCII-only type name of `T`.
184///
185/// For example:
186/// - `my_crate::foo::Bar` -> `"Bar"`
187/// - `my_crate::foo::Café` -> `"Caf"`
188///
189/// Non-ASCII characters are stripped so generated internal identifiers remain
190/// stable, predictable, and compiler-friendly.
191///
192/// This is useful when generating proc-macro identifiers or diagnostics
193/// without including module paths or Unicode normalization concerns.
194fn bare_type_name<T: 'static>() -> String {
195    use std::any::type_name;
196
197    let k: String = type_name::<T>()
198        .rsplit("::")
199        .next()
200        .unwrap()
201        .chars()
202        .filter(|c| c.is_ascii())
203        .collect();
204
205    k.replace("::", "_")
206        .replace(['<', '>'], "_")
207        .chars()
208        .filter(|c| c.is_ascii())
209        .collect()
210}

212/// Generates a deterministic, type-scoped identifier for use inside
213/// proc-macro expansions.
214///
215/// **Why this exists:**
216/// Proc macros often need to invent identifiers that are unlikely to collide
217/// with user code. This function derives the identifier from the type's
218/// textual name rather than relying on runtime state or a generated counter.
219///
220/// **How it works:**
221/// The generic parameter `T` is never constructed or used as a value.
222/// Its fully qualified type name is obtained using [`std::any::type_name`]
223/// and used as the identity anchor for identifier generation.
224///
225/// Because the complete type name is used as the anchor, different types
226/// normally produce different generated identifiers while repeated requests
227/// for the same type produce the same identifier.
228///
229/// The generated identifier is formatted according to the requested casing,
230/// with optional prefix and suffix applied.
231fn gen_ident_from_marker<T: 'static>(
232    case: Case,
233    prefix: Option<&[u8]>,
234    suffix: Option<&[u8]>,
235) -> String {
236    let key = bare_type_name::<T>();
237
238    apply_case(&key, case, prefix, suffix)
239}

241/// Generates a deterministic, identifier-scoped identifier for use inside
242/// proc-macro expansions.
243///
244/// **Why this exists:**
245/// Some proc-macro scenarios already have a meaningful identifier available
246/// and need to derive additional symbols from its name without introducing
247/// a dedicated marker type.
248///
249/// **How it works:**
250/// The provided [`Ident`] is converted to its textual representation.
251/// Span and hygiene information are deliberately ignored, making the
252/// identifier name itself the identity anchor.
253///
254/// The same identifier name therefore produces the same generated identifier
255/// regardless of its span or hygiene context.
256///
257/// The generated identifier is formatted according to the requested casing,
258/// with optional prefix and suffix applied.
259fn gen_ident_from_ident(
260    ident: &Ident,
261    case: Case,
262    prefix: Option<&[u8]>,
263    suffix: Option<&[u8]>,
264) -> String {
265
266    let key = ident.to_string();
267
268    apply_case(&key, case, prefix, suffix)
269}

271/// Generates a deterministic, token-scoped identifier for use inside
272/// proc-macro expansions.
273///
274/// **Why this exists:**
275/// Some proc-macro scenarios need to derive identifiers directly from syntax
276/// rather than from a marker type or an existing identifier. In those cases,
277/// the token stream itself provides the identity anchor.
278///
279/// **How it works:**
280/// The provided [`TokenStream`] is converted into its textual representation
281/// and used as the identity anchor for identifier generation.
282///
283/// Equivalent token-stream text therefore produces the same generated
284/// identifier without requiring shared marker types, identifier names,
285/// or runtime state.
286///
287/// The generated identifier is formatted according to the requested casing,
288/// with optional prefix and suffix applied.
289fn gen_ident_from_tokens(
290    tokens: &TokenStream,
291    case: Case,
292    prefix: Option<&[u8]>,
293    suffix: Option<&[u8]>,
294) -> String {
295    let key = tokens.to_string();
296
297    apply_case(&key, case, prefix, suffix)
298}

300/// Applies a casing style and optional prefix/suffix to a base identifier.
301///
302/// **Why this exists:**  
303/// Identifier generation in proc macros needs to be predictable, hygienic,
304/// and fast. This function centralizes all string-shaping logic so the rest
305/// of the codebase can focus purely on *what* identifier is being generated,
306/// not *how* it is formatted.
307///
308/// **What it does:**  
309/// - Takes a base identifier word (`word`)
310/// - Optionally prepends and/or appends byte-based prefixes and suffixes
311/// - Formats the result according to the requested casing style:
312///   - `Lower`: lowercase alphanumeric segments separated by `_`
313///   - `Upper`: uppercase alphanumeric segments separated by `_`
314///   - `Pascal`: PascalCase with non-alphanumeric characters treated as
315///     word boundaries
316///   - Always prepends __ to the resulting identifier.
317///
318/// Non-alphanumeric characters are treated as separators:
319/// - `Lower` and `Upper` collapse each separator run into a single `_`.
320/// - `Pascal` removes separators and capitalizes the next alphanumeric
321///   character.
322///
323/// Prefixes and suffixes are provided as byte slices to keep internal
324/// identifier components lightweight and avoid unnecessary allocations.
325pub(crate) fn apply_case(
326    word: &str,
327    case: Case,
328    prefix: Option<&[u8]>,
329    suffix: Option<&[u8]>,
330) -> String {
331    let mut out = String::with_capacity(
332        word.len()
333            + prefix.map_or(0, |p| p.len() + 1)
334            + suffix.map_or(0, |s| s.len() + 1),
335    );
336
337    match case {
338        Case::Lower => {
339            append_lower(&mut out, prefix);
340            append_separator(&mut out);
341            append_lower_str(&mut out, word);
342            append_separator(&mut out);
343            append_lower(&mut out, suffix);
344
345            trim_separator(&mut out);
346        }
347
348        Case::Upper => {
349            append_upper(&mut out, prefix);
350            append_separator(&mut out);
351            append_upper_str(&mut out, word);
352            append_separator(&mut out);
353            append_upper(&mut out, suffix);
354
355            trim_separator(&mut out);
356        }
357
358        Case::Pascal => {
359            if let Some(prefix) = prefix {
360                push_pascal_bytes(&mut out, prefix);
361            }
362
363            push_pascal_str(&mut out, word);
364
365            if let Some(suffix) = suffix {
366                push_pascal_bytes(&mut out, suffix);
367            }
368        }
369    }
370
371    out.insert_str(0, "__");
372
373    out
374}

376/// Appends a lowercase transformation of an ASCII byte slice.
377///
378/// Non-alphanumeric characters are treated as word boundaries and collapsed
379/// into a single underscore when followed by an alphanumeric character.
380fn append_lower(out: &mut String, bytes: Option<&[u8]>) {
381    let Some(bytes) = bytes else {
382        return;
383    };
384
385    let mut separator = false;
386
387    for &b in bytes {
388        if b.is_ascii_alphanumeric() {
389            if separator && !out.is_empty() && !out.ends_with('_') {
390                out.push('_');
391            }
392
393            out.push(b.to_ascii_lowercase() as char);
394            separator = false;
395        } else {
396            separator = true;
397        }
398    }
399}

468/// Appends a PascalCase transformation of an ASCII byte slice.
469///
470/// Non-alphanumeric characters are treated as word boundaries and removed.
471/// The next alphanumeric character is capitalized.
472fn push_pascal_bytes(out: &mut String, bytes: &[u8]) {
473    let mut capitalize_next = true;
474
475    for &b in bytes {
476        if !b.is_ascii_alphanumeric() {
477            capitalize_next = true;
478            continue;
479        }
480
481        let c = b as char;
482
483        if capitalize_next {
484            out.push(c.to_ascii_uppercase());
485            capitalize_next = false;
486        } else {
487            out.push(c);
488        }
489    }
490}
491
492/// Appends a PascalCase transformation of a string slice.
493///
494/// Non-alphanumeric characters are treated as word boundaries and removed.
495/// The next alphanumeric character is capitalized.
496fn push_pascal_str(out: &mut String, s: &str) {
497    let mut capitalize_next = true;
498
499    for c in s.chars() {
500        if !c.is_ascii_alphanumeric() {
501            capitalize_next = true;
502            continue;
503        }
504
505        if capitalize_next {
506            out.push(c.to_ascii_uppercase());
507            capitalize_next = false;
508        } else {
509            out.push(c);
510        }
511    }
512}

514/// Appends a separator when the output is non-empty and does not already end
515/// with one.
516fn append_separator(out: &mut String) {
517    if !out.is_empty() && !out.ends_with('_') {
518        out.push('_');
519    }
520}
521
522/// Removes trailing separators from the generated identifier.
523fn trim_separator(out: &mut String) {
524    while out.ends_with('_') {
525        out.pop();
526    }
