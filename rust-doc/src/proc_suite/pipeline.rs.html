38/// Generates a structured proc-macro pipeline as a set of local traits.
39///
40/// Procedural macros operate on syntax represented as `TokenStream`s.
41/// Input is parsed into structured syntax, analyzed, transformed, and
42/// ultimately emitted back as a `TokenStream`.
43///
44/// This macro organizes that process into explicit pipeline stages:
45///
46/// `text
47/// Extract -> Extend -> Transform -> Insert
48///      \
49///       -> Utilize
50/// `
51///
52/// - **Extract**: derives owned, pipeline-friendly structures from syntax.
53/// - **Utilize**: derives borrowed, lifetime-carrying views over existing syntax.
54/// - **Extend**: conditionally derives and inserts new structures.
55/// - **Transform**: mutates existing structures in place.
56/// - **Insert**: integrates data into a target representation.
57///
58/// Procedural macros are fundamentally syntactic transformations. Some
59/// workflows benefit from extracting syntax into owned intermediate
60/// representations, while others can operate directly on borrowed views
61/// of existing syntax.
62///
63/// The stages can therefore be composed according to the transformation:
64///
65/// - `extract -> extend -> insert`
66/// - `extract -> transform`
67/// - `utilize -> transform`
68/// - `utilize -> extend -> insert`
69/// - or any combination of stages.
70///
71/// Each stage supports two execution modes:
72///
73/// - **checked**: performs pre- and post-validation.
74/// - **unchecked**: performs minimal execution.
75///
76/// ## Syntax
77///
78/// ```ignore
79/// proc_pipeline! {
80///     struct MacroGlobalIdent {
81///         extract: ExtractTraitIdent,
82///         insert: InsertTraitIdent,
83///         extend: ExtendTraitIdent,
84///         transform: TransformTraitIdent,
85///         utilize: UtilizeTraitIdent,
86///         support: "support_crate",
87///         delegate: ("delegate_crate", "delegate_macro"), // optional
88///     }
89/// }
90/// ```
91///
92/// `support` specifies the support crate used by generated code.
93/// `delegate` **optionally** specifies the delegated support crate and its
94/// delegate attribute macro.
95///
96/// See [`SupportCrate`] for the semantics of support-crate resolution
97/// and delegation.
98///
99/// ## Generated Traits
100///
101/// The macro expands into traits declared at the invocation site
102/// (`pub(crate)`). This allows implementations for foreign types, such as
103/// `syn` syntax nodes, while avoiding orphan-rule restrictions by keeping
104/// the traits local to the crate.
105///
106/// ## Purpose
107///
108/// Provides a predictable and composable structure for proc-macro logic,
109/// replacing ad-hoc transformations with explicit pipeline stages and
110/// defined ownership boundaries between extracted and utilizable
111/// representations.
112#[macro_export]
113macro_rules! proc_pipeline {
114
115    (@delegate $Delegate:literal, $DelegateProcMacro:literal) => {
116        fn delegate_crate() -> Option<syn::Ident> {
117            Some(::quote::format_ident!($Delegate))
118        }
119
120        fn delegate_macro() -> Option<syn::Ident> {
121            Some(::quote::format_ident!($DelegateProcMacro))
122        }
123    };
124
125    (@delegate) => {};
126
127    (
128        $(#[$Docs:meta])*
129        $Vis:vis struct $Type:ident {
130            extract: $Extract:ident,
131            insert: $Insert:ident,
132            extend: $Extend:ident,
133            transform: $Transform:ident,
134            utilize: $Utilize:ident,
135            support: $Support:literal
136            $(, delegate: ($Delegate:literal, $DelegateProcMacro:literal))?
137            $(,)?
138        }
139    ) => {
140
141        $(#[$Docs])*
142        $Vis struct $Type;
143
144        impl $crate::pipeline::SupportCrate for $Type {
145            fn support_crate() -> syn::Ident {
146                quote::format_ident!($Support)
147            }
148
149            proc_pipeline!(@delegate
150                $($Delegate, $DelegateProcMacro)?
151            );
152        }

154        /// Defines extraction of structured data from an input source.
155        ///
156        /// This trait represents the initial stage of a structured proc-macro
157        /// pipeline, where relevant information is derived from syntax or other
158        /// inputs and prepared for further processing.
159        ///
160        /// `From` typically represents the syntax node (or source) from which
161        /// `Self` is derived, under the given `Context`.
162        ///
163        /// `Context` provides optional external state, configuration, or borrowed
164        /// supporting data used during extraction.
165        ///
166        /// Since `Self` is typically owned (`'static`), `Context` is commonly used
167        /// to provide temporary borrowed data required during derivation or validation.
168        ///
169        /// This trait is usually implemented for types that hold the extracted
170        /// data used in later processing stages.
171        ///
172        /// Provides two execution paths:
173        /// - [`Self::checked_extract`]: includes validation before and after extraction
174        /// - [`Self::unchecked_extract`]: performs extraction with minimal checks
175        ///
176        /// Implementors control input validation, extraction logic, and
177        /// post-extraction verification.
178        pub(crate) trait $Extract<From, Context = ()>
179        where
180            Self: Sized + 'static + Clone + std::fmt::Debug,
181        {
182            /// Performs extraction with full validation.
183            ///
184            /// Validates the source and context, executes extraction,
185            /// and verifies the resulting value.
186            fn checked_extract(
187                from: &From,
188                context: &Context,
189            ) -> Result<Self, proc_macro2::TokenStream> {
190                Self::validate_from(from)?;
191                Self::validate_context(context)?;
192                let result = Self::raw_extract(from, context)?;
193                result.validate_extract(from, Some(context))?;
194                Ok(result)
195            }
196
197            /// Performs extraction with minimal validation.
198            ///
199            /// Skips post-extraction validation, returning the raw result.
200            fn unchecked_extract(
201                from: &From,
202                context: &Context,
203            ) -> Result<Self, proc_macro2::TokenStream> {
204                Self::validate_from(from)?;
205                Self::validate_context(context)?;
206                Self::raw_extract(from, context)
207            }
208
209            /// Validates the input source before extraction.
210            ///
211            /// Override to enforce preconditions on `From`.
212            ///
213            /// Default is no-op.
214            fn validate_from(_from: &From) -> Result<(), proc_macro2::TokenStream> {
215                Ok(())
216            }
217
218            /// Validates the extraction context before extraction.
219            ///
220            /// Override to enforce constraints on `Context`.
221            ///
222            /// Default is no-op.
223            fn validate_context(_context: &Context) -> Result<(), proc_macro2::TokenStream> {
224                Ok(())
225            }
226
227            /// Performs the core extraction logic.
228            ///
229            /// Converts `From` into `Self` using the given `Context`.
230            fn raw_extract(from: &From, context: &Context) -> Result<Self, proc_macro2::TokenStream>;
231
232            /// Validates the extracted result.
233            ///
234            /// `context` is optional so this validation hook may be reused
235            /// independently from the checked extraction pipeline, such as:
236            ///
237            /// - standalone tests,
238            /// - partial pipeline execution,
239            /// - manual validation,
240            /// - or external validation utilities.
241            fn validate_extract(
242                &self,
243                from: &From,
244                context: Option<&Context>,
245            ) -> Result<(), proc_macro2::TokenStream>;
246        }
247
248        /// Defines insertion of `Self` into a target type (`Into`) using a given context.
249        ///
250        /// This represents the mutation stage of a structured proc-macro pipeline,
251        /// where `Self` is integrated into `Into` as part of its final structure.
252        ///
253        /// `Into` is typically a syntax node or target representation, while
254        /// `Context` provides external rules or configuration required during
255        /// insertion (such as naming policy, validation rules, feature flags,
256        /// or expansion constraints).
257        ///
258        /// `Context` provides optional external state, configuration, or borrowed
259        /// supporting data used during insertion.
260        ///
261        /// Since `Self` is typically owned (`'static`), `Context` is commonly used
262        /// to provide temporary borrowed data required during derivation or validation.
263        ///
264        /// `Self` may represent:
265        /// - a generated syntax node,
266        /// - extracted metadata being re-inserted,
267        /// - or contextual data projected into the target representation.
268        ///
269        /// Provides two execution paths:
270        /// - [`Self::checked_insert`] performs full validation before and after insertion
271        /// - [`Self::unchecked_insert`] performs insertion with minimal checks
272        ///
273        /// This keeps insertion predictable, validated, and composable within the
274        /// larger proc-macro pipeline.
275        pub(crate) trait $Insert<Into, Context = ()>
276        where
277            Self: Sized + 'static + Clone + std::fmt::Debug,
278        {
279            /// Performs insertion with full validation.
280            ///
281            /// Validates the insertion context and target, applies the insertion,
282            /// and verifies the final inserted state.
283            fn checked_insert(
284                &self,
285                to: &mut Into,
286                context: &Context,
287            ) -> Result<(), proc_macro2::TokenStream> {
288                self.validate_context(context)?;
289                self.validate_into(to, context)?;
290                self.raw_insert(to, context)?;
291                Self::validate_inserted(Some(self), to, Some(context))
292            }
293
294            /// Performs insertion with minimal validation.
295            ///
296            /// Validates preconditions and applies insertion without checking
297            /// the final inserted state.
298            fn unchecked_insert(
299                &self,
300                to: &mut Into,
301                context: &Context,
302            ) -> Result<(), proc_macro2::TokenStream> {
303                self.validate_context(context)?;
304                self.validate_into(to, context)?;
305                self.raw_insert(to, context)
306            }
307
308            /// Validates the insertion context before insertion.
309            ///
310            /// Override to enforce constraints on the external insertion context.
311            ///
312            /// Default implementation performs no validation.
313            fn validate_context(&self, _context: &Context) -> Result<(), proc_macro2::TokenStream> {
314                Ok(())
315            }
316
317            /// Validates the target before insertion.
318            ///
319            /// Override to ensure the target is in a valid state to receive
320            /// the insertion.
321            ///
322            /// Default implementation performs no validation.
323            fn validate_into(
324                &self,
325                _to: &Into,
326                _context: &Context,
327            ) -> Result<(), proc_macro2::TokenStream> {
328                Ok(())
329            }
330
331            /// Performs the core insertion logic.
332            ///
333            /// Integrates `Self` into `Into`, mutating the target representation
334            /// using the provided context.
335            fn raw_insert(
336                &self,
337                to: &mut Into,
338                context: &Context,
339            ) -> Result<(), proc_macro2::TokenStream>;
340
341            /// Validates the target after insertion.
342            /// to ensure the resulting structure remains valid.
343            ///
344            /// `context` and `inserted`` are optional so this validation hook may be reused
345            /// independently from the checked extraction pipeline, such as:
346            ///
347            /// - standalone tests,
348            /// - partial pipeline execution,
349            /// - manual validation,
350            /// - or external validation utilities.
351            fn validate_inserted(
352                _of: Option<&Self>,
353                _to: &Into,
354                _context: Option<&Context>,
355            ) -> Result<(), proc_macro2::TokenStream>;
356        }
357
358        /// Defines extension of a target (`Towards`) within the proc-macro pipeline.
359        ///
360        /// This stage sits between extraction and insertion:
361        #[doc = concat!(
362                    "- extraction derives structured data via [`", stringify!($Extract), "`],\n",
363                    "- insertion integrates generated data via [`", stringify!($Insert), "`],"
364                )]
365        /// - extension uses extracted or contextual data (`Self`) to derive a new
366        ///   `Type`, which is then inserted into `Towards`.
367        ///
368        /// `Self` usually represents validated extracted data or contextual state
369        /// that decides *what* should be added.
370        ///
371        /// `Context` provides optional external state, configuration, or borrowed
372        /// supporting data used during extension.
373        ///
374        /// Since `Self` is typically owned (`'static`), `Context` is commonly used
375        /// to provide temporary borrowed data required during derivation or validation.
376        ///
377        /// The produced `Type` must implement
378        #[doc = concat!("[`", stringify!($Insert), "`],")]
379        /// allowing it to be inserted into the target.
380        ///
381        /// Provides two execution paths:
382        /// - [`Self::checked_extend`] performs validation before derivation and insertion
383        /// - [`Self::unchecked_extend`] performs extension with minimal checks
384        ///
385        /// This enables controlled structural growth of syntax nodes and target
386        /// representations during macro expansion.
387        pub(crate) trait $Extend<Type, Towards, Context = ()>
388        where
389            Self: Sized + 'static + Clone + std::fmt::Debug,
390            Type: $Insert<Towards, Context> + Sized,
391        {
392            /// Performs extension with full validation.
393            ///
394            /// Validates the context and target, derives the item,
395            /// validates the derived result, and inserts it using
396            #[doc = concat!("[`", stringify!($Insert), "`].")]
397            fn checked_extend(
398                &self,
399                insert_to: &mut Towards,
400                context: &Context,
401            ) -> Result<(), proc_macro2::TokenStream> {
402                self.validate_context(context)?;
403                self.validate_towards(insert_to, context)?;
404                let item = self.raw_extend(insert_to, context)?;
405                item.checked_insert(insert_to, context)?;
406                self.validate_extend(Some(&item), insert_to, Some(context))?;
407                Ok(())
408            }
409
410            /// Performs extension with minimal validation.
411            ///
412            /// Validates preconditions, derives the item, and inserts it
413            /// without validating the derived value.
414            fn unchecked_extend(
415                &self,
416                insert_to: &mut Towards,
417                context: &Context,
418            ) -> Result<(), proc_macro2::TokenStream> {
419                self.validate_context(context)?;
420                self.validate_towards(insert_to, context)?;
421                let item = self.raw_extend(insert_to, context)?;
422                item.unchecked_insert(insert_to, context)?;
423                Ok(())
424            }
425
426            /// Validates the extension context before extension.
427            ///
428            /// Override to enforce constraints on the external extension context.
429            ///
430            /// Default implementation performs no validation.
431            fn validate_context(&self, _context: &Context) -> Result<(), proc_macro2::TokenStream> {
432                Ok(())
433            }
434
435            /// Validates the target before extension.
436            ///
437            /// Override to ensure the target is in a valid state to receive
438            /// the derived insertion.
439            ///
440            /// Default implementation performs no validation.
441            fn validate_towards(
442                &self,
443                _towards: &Towards,
444                _context: &Context,
445            ) -> Result<(), proc_macro2::TokenStream> {
446                Ok(())
447            }
448
449            /// Produces the item to be inserted into the target.
450            ///
451            /// Derives a new `Type` from `Self` using the current state
452            /// of `Towards` and the provided context.
453            ///
454            /// The returned value is later inserted via
455            #[doc = concat!("[`", stringify!($Insert), "`].")]
456            fn raw_extend(
457                &self,
458                towards: &Towards,
459                context: &Context,
460            ) -> Result<Type, proc_macro2::TokenStream>;
461
462            /// Validates the extended result
463            ///
464            /// `context` and `item` are optional so this validation hook may be reused
465            /// independently from the checked extraction pipeline, such as:
466            ///
467            /// - standalone tests,
468            /// - partial pipeline execution,
469            /// - manual validation,
470            /// - or external validation utilities.
471            fn validate_extend(
472                &self,
473                item: Option<&Type>,
474                towards: &Towards,
475                context: Option<&Context>,
476            ) -> Result<(), proc_macro2::TokenStream>;
477        }

479        /// Defines transformation of an existing value (`Type`) in place.
480        ///
481        /// Unlike extension (which derives and inserts new data), transformation
482        /// operates directly on the target, modifying or projecting its internal
483        /// representation.
484        #[doc = concat!(
485                    "See [`", stringify!($Extend), "`] for extension-based workflows."
486                )]
487        ///
488        /// `Self` acts as contextual or extracted data used to drive how the
489        /// transformation is applied.
490        ///
491        /// `Context` provides optional external state, configuration, or borrowed
492        /// supporting data used during transformation.
493        ///
494        /// Since `Self` is typically owned (`'static`), `Context` is commonly used
495        /// to provide temporary borrowed data required during derivation or validation.
496        ///
497        /// Provides two execution paths:
498        /// - [`Self::checked_transform`]: applies transformation with post-validation
499        /// - [`Self::unchecked_transform`]: applies transformation with minimal checks
500        ///
501        /// This is typically used for rewriting, normalizing, or updating a
502        /// structure without introducing new elements via insertion.
503        pub(crate) trait $Transform<Type, Context = ()>
504        where
505            Self: Sized + 'static + Clone + std::fmt::Debug,
506        {
507            /// Performs transformation with full validation.
508            ///
509            /// Checks preconditions, applies the transformation,
510            /// and validates the resulting state.
511            fn checked_transform(&self, transform: &mut Type, context: &Context) -> Result<(), proc_macro2::TokenStream> {
512                self.validate_type(transform)?;
513                self.validate_context(context)?;
514                self.raw_transform(transform, context)?;
515                self.validate_transform(transform, Some(context))?;
516                Ok(())
517            }
518
519            /// Performs transformation with minimal validation.
520            ///
521            /// Applies the transformation without validating the result.
522            fn unchecked_transform(&self, transform: &mut Type, context: &Context) -> Result<(), proc_macro2::TokenStream> {
523                self.validate_type(transform)?;
524                self.validate_context(context)?;
525                self.raw_transform(transform, context)?;
526                Ok(())
527            }
528
529
530            /// Validates the transformation context.
531            ///
532            /// Default implementation performs no validation.
533            fn validate_context(&self, _context: &Context) -> Result<(), proc_macro2::TokenStream> {
534                Ok(())
535            }
536
537            /// Validates the target before transformation.
538            ///
539            /// Override to ensure the target is in a valid state prior
540            /// to mutation.
541            ///
542            /// Default implementation performs no validation.
543            fn validate_type(
544                &self,
545                _type: &Type,
546            ) -> Result<(), proc_macro2::TokenStream> {
547                Ok(())
548            }
549
550            /// Performs the core transformation logic.
551            ///
552            /// Mutates the target in place using `Self` as context,
553            /// updating its internal representation.
554            fn raw_transform(&self, transform: &mut Type, context: &Context) -> Result<(), proc_macro2::TokenStream>;
555
556            /// Validates the transformed state.
557            ///
558            /// `context` and `type` are optional so this validation hook may be reused
559            /// independently from the checked extraction pipeline, such as:
560            ///
561            /// - standalone tests,
562            /// - partial pipeline execution,
563            /// - manual validation,
564            /// - or external validation utilities.
565            fn validate_transform(&self, transform: &Type, context: Option<&Context>) -> Result<(), proc_macro2::TokenStream>;
566        }

569        /// Defines derivation of data from an input source without requiring
570        /// ownership of the resulting representation.
571        ///
572        /// Procedural macros are fundamentally syntactic transformations.
573        /// While some workflows benefit from extracting syntax into owned,
574        /// pipeline-friendly structures, many transformations primarily operate
575        /// on existing syntax trees and reuse most of their original structure.
576        ///
577        /// In such cases, constructing a fully owned extraction model may be
578        /// unnecessary. Borrowing from the source syntax can more naturally
579        /// express the relationship between derived data and the structures from
580        /// which it originates while avoiding allocation of owned intermediates.
581        ///
582        /// Unlike
583        #[doc = concat!("[`", stringify!($Extract), "`],")]
584        /// this trait is designed for lifetime-carrying representations.
585        /// Implementors are permitted to borrow from the source and context,
586        /// allowing `Self` to act as a lightweight utilizable view over existing
587        /// syntax rather than an owned pipeline artifact.
588        ///
589        /// `From` typically represents the syntax node (or source) from which
590        /// `Self` is derived under the given `Context`.
591        ///
592        /// `Context` provides optional external state, configuration, or borrowed
593        /// supporting data used during derivation.
594        ///
595        /// The lifetime `'a` represents the lifetime of data borrowed from the
596        /// source and context. Implementations may use it to construct temporary
597        /// projections, views, or metadata representations tied to the lifetime
598        /// of the originating structures.
599        ///
600        /// This trait is commonly used when:
601        ///
602        /// - inspecting existing syntax without constructing owned intermediates,
603        /// - borrowing directly from syntax trees,
604        /// - building temporary projections or views,
605        /// - sharing references to contextual information,
606        /// - or driving transformations from existing structures.
607        ///
608        /// Provides two execution paths:
609        ///
610        /// - [`Self::checked_utilize`]: includes validation before and after derivation
611        /// - [`Self::unchecked_utilize`]: performs derivation with minimal checks
612        ///
613        /// Implementors control source validation, derivation logic, and
614        /// post-derivation verification.
615        ///
616        /// See
617        #[doc = concat!("[`", stringify!($Extract), "`]")]
618        /// when an owned (`'static`) pipeline artifact is required.
619        pub(crate) trait $Utilize<'a, From, Context = ()>
620        where
621            Self: Sized + std::fmt::Debug + 'a,
622        {
623
624            /// Performs derivation of a utilizable value with full validation.
625            ///
626            /// Validates the source and context, constructs the utilizable value,
627            /// and verifies the resulting representation.
628            fn checked_utilize(
629                from: &'a From,
630                context: &'a Context,
631            ) -> Result<Self, proc_macro2::TokenStream> {
632                Self::validate_from(from)?;
633                Self::validate_context(context)?;
634                let result = Self::raw_utilize(from, context)?;
635                result.validate_utilize(from, Some(context))?;
636                Ok(result)
637            }
638
639
640            /// Performs derivation of a utilizable value with minimal validation.
641            ///
642            /// Skips post-derivation validation, returning the raw result.
643            fn unchecked_utilize(
644                from: &'a From,
645                context: &'a Context,
646            ) -> Result<Self, proc_macro2::TokenStream> {
647                Self::validate_from(from)?;
648                Self::validate_context(context)?;
649                Self::raw_utilize(from, context)
650            }
651
652
653            /// Validates the input source before construction of the utilizable value.
654            ///
655            /// Override to enforce preconditions on `From`.
656            ///
657            /// Default implementation performs no validation.
658            fn validate_from(_from: &From) -> Result<(), proc_macro2::TokenStream> {
659                Ok(())
660            }
661
662
663            /// Validates the utilization context before construction of the
664            /// utilizable value.
665            ///
666            /// Override to enforce constraints on `Context`.
667            ///
668            /// Default implementation performs no validation.
669            fn validate_context(_context: &Context) -> Result<(), proc_macro2::TokenStream> {
670                Ok(())
671            }
672
673            /// Performs the core utilization logic.
674            ///
675            /// Constructs a utilizable representation from `From` using the given
676            /// `Context`.
677            ///
678            /// Unlike owned extraction pipelines, implementations may return
679            /// borrowed, lifetime-carrying, or otherwise non-`'static` views over
680            /// the source representation.
681            fn raw_utilize(from: &'a From, context: &'a Context) -> Result<Self, proc_macro2::TokenStream>;
682
683            /// Validates the utilizable result.
684            ///
685            /// `context` is optional so this validation hook may be reused
686            /// independently from the checked utilization pipeline, such as:
687            ///
688            /// - standalone tests,
689            /// - partial pipeline execution,
690            /// - manual validation,
691            /// - or external validation utilities.
692            fn validate_utilize(
693                &self,
694                from: &From,
695                context: Option<&Context>,
696            ) -> Result<(), proc_macro2::TokenStream>;
697        }
698
699    };

702/// Describes the crate-resolution and delegation model used by a proc-macro
703/// pipeline.
704///
705/// # Why a support crate is necessary
706///
707/// A procedural macro is implemented in a proc-macro crate, but the code
708/// produced by that macro is ultimately compiled in the user's crate. The
709/// generated code may need to refer to traits, types, helper macros, or other
710/// items that belong to the library associated with the proc macro.
711///
712/// Those items should be accessed through a normal library crate rather than
713/// by making generated code depend directly on the proc-macro implementation
714/// crate. The normal library crate acts as the public support layer:
715///
716/// ```text
717/// user crate
718///     |
719///     |--- depends on -> support crate
720///                           |
721///                           |--- depends on -> proc-macro crate
722/// ```
723///
724/// For example, suppose a library provides a proc macro and also provides a
725/// trait required by the code generated by that macro:
726///
727/// ```text
728/// my_library
729///     |--- public support items
730///     |     |--- Trait
731///     |     |--- Type
732///     |
733///     |--- re-exports -> my_library_macro
734/// ```
735///
736/// A user then depends on `my_library`, not directly on
737/// `my_library_macro`. If the proc macro generates an implementation of
738/// `Trait`, the generated code should therefore refer to the trait through
739/// the public library:
740///
741/// ```ignore
742/// my_library::Trait
743/// ```
744///
745/// rather than directly referring to the implementation crate.
746///
747/// The support crate can re-export the proc-macro entry points and the items
748/// required by generated code. This gives the generated code one stable,
749/// user-visible crate through which it can access everything belonging to the
750/// macro's public interface.
751///
752/// Consequently, generated code commonly needs to construct paths whose first
753/// segment is the identifier returned by [`Self::support_crate`]:
754///
755/// ```ignore
756/// #support_crate::Trait
757/// #support_crate::Type
758/// #support_crate::helper_macro!
759/// ```
760///
761/// For example, if `support_crate()` returns the identifier `my_library`, the
762/// generated paths become conceptually:
763///
764/// ```ignore
765/// my_library::Trait
766/// my_library::Type
767/// my_library::helper_macro!
768/// ```
769///
770/// [`Self::support_crate`] provides the identifier used as the first path
771/// segment for these generated paths. Keeping this information in the
772/// pipeline makes crate resolution explicit and allows every part of the
773/// pipeline to construct paths consistently.
774///
775/// # Why delegation is necessary
776///
777/// A proc macro can itself be used as a building block by another proc macro.
778/// In that situation, the inner proc macro still has its own support crate,
779/// but the user may depend only on the outer macro's support crate.
780///
781/// The inner support crate can re-export the outer support crate's public
782/// items. This means the generated code from the outer macro must sometimes
783/// refer to its own outer support crate rather than directly to the inner macro's
784/// original support crate.
785///
786/// For example, consider two layers of proc-macro composition:
787///
788/// ```text
789/// user crate
790///     |
791///     |--- depends on -> inner support crate
792///                           |
793///                           |--- re-exports inner support items
794///                           |
795///                           |--- uses -> inner proc macro
796/// ```
797///
798/// The inner proc macro may generate:
799///
800/// ```ignore
801/// inner_support::Trait
802/// ```
803///
804/// but the composed (nested) macro needs that reference to resolve through the outer
805/// support crate:
806///
807/// ```ignore
808/// outer_support::Trait
809/// ```
810///
811/// The inner macro does not necessarily know that it is being used through
812/// the outer layer. Delegation provides the mechanism by which the inner
813/// pipeline can communicate that new crate-resolution context to the outer
814/// generated syntax.
815///
816/// This creates a simple crate-identifier substitution:
817///
818/// ```text
819/// delegated support crate
820///         |
821///         | delegation
822///         |
823/// support crate
824/// ```
825///
826/// [`Self::delegate_crate`] identifies the crate whose references should be
827/// replaced by the inner macro's [`Self::support_crate`] when such delegation
828/// is required.
829///
830/// The replacement is performed by [`Self::delegate_inner`], which operates
831/// directly on a [`syn`] syntax representation and replaces occurrences of the
832/// delegated crate identifier with the support-crate identifier.
833///
834/// For example, conceptually:
835///
836/// before delegation:
837/// ```ignore
838/// outer_support::Trait
839/// outer_support::Type
840/// ```
841/// after delegation:
842/// ```ignore
843/// inner_support::Trait
844/// inner_support::Type
845/// ```
846///
847/// Delegation is optional. A pipeline that is not being embedded inside
848/// another proc-macro pipeline does not need to perform any replacement, so
849/// the default implementation returns `None`.
850///
851/// # Why a delegate macro is required for attributes
852///
853/// Function-like and attribute procedural macros differ in how their input
854/// and generated output participate in macro expansion.
855///
856/// With a function-like proc macro, an outer proc macro can construct another
857/// macro invocation around its input. The inner macro invocation can
858/// therefore be represented explicitly in the tokens being passed through
859/// the expansion process. This makes it possible for the outer pipeline to
860/// arrange for the inner expansion and subsequently rewrite the resulting
861/// syntax.
862///
863/// For example, an outer macro can conceptually produce:
864///
865/// ```ignore
866/// // here, we describe outer and inner as structures not
867/// // dependencies
868///
869/// outer_macro! {
870///     { "a user given literal expr example" }
871/// }
872///
873/// outer_macro_of_inner! {
874///     inner_macro! {
875///         { "a user given literal expr example" }
876///     }
877/// }
878///
879/// outer_macro_of_inner_input!{
880///      { "a inner macro transformed literal" }
881/// }
882/// ```
883///
884/// The inner macro expansion can then become part of the syntax that the
885/// outer pipeline can work with.
886///
887/// Attribute macros have a different expansion relationship. An outer
888/// attribute macro cannot inspect syntax that will only be generated later
889/// by an inner attribute macro. In particular, an outer transformation cannot
890/// retroactively modify the output of an inner attribute expansion merely by
891/// rewriting the syntax it originally received.
892///
893/// For example, consider:
894///
895/// ```ignore
896/// #[outer]
897/// #[inner]
898/// impl Sum for Marker { ... };
899/// ```
900///
901/// The outer attribute receives the item as it exists at that point in the
902/// expansion process. It cannot simply rewrite the syntax that `inner` will
903/// produce later, because that generated syntax does not yet exist from the
904/// outer macro's point of view.
905///
906/// Therefore, attribute delegation requires a second mechanism: a delegate
907/// attribute that is attached to the input (pushed to the bottom)
908/// which invokes after the inner attribute macro expands.
909///
910/// Conceptually, the expansion proceeds as:
911///
912/// ```text
913/// original item
914///             |
915///       outer macro expands - adds delegate attribute
916///             |
917///       inner macro expands
918///             |
919///       delegate attribute runs
920///             |
921///       support-crate references are rewritten
922/// ```
923///
924/// For example, the pipeline can arrange the attributes conceptually as:
925///
926/// ```ignore
927/// #[inner_support::outer_macro] // since outer macros are re-expoted to inner support crate
928/// #[inner_support::delegate_macro]
929/// impl Sum for Marker { ... };
930/// ```
931///
932/// The inner macro expands first and may produce:
933///
934/// ```ignore
935/// outer_support::Trait
936/// outer_support::Type
937/// ```
938///
939/// The delegate attribute then receives that resulting syntax and rewrites
940/// those references to:
941///
942/// ```ignore
943/// inner_support::Trait
944/// inner_support::Type
945/// ```
946///
947/// The important point is that the delegate attribute does not perform the
948/// inner transformation itself. Its purpose is to run at the correct point
949/// in the expansion process so that the syntax produced by the re-exported
950/// outer macro is available for path rewriting.
951///
952/// [`Self::delegate_macro`] provides the identifier of this delegate
953/// attribute. The delegate macro is expected to perform the support-crate
954/// replacement after the outer macro has produced its syntax.
955///
956/// [`Self::delegate_attribute`] attaches this delegate attribute to a syntax
957/// item. The attribute is placed at the end of the existing attribute list so
958/// that it can operate after the relevant outer macro expansion.
959///
960/// Both the delegate crate and delegate macro are optional. If either is not
961/// configured, no delegate attribute is added.
962///
963/// # Direct delegation
964///
965/// [`Self::delegate_item`] is the direct form of delegation. It is used when
966/// the current macro has access to the inner syntax (of outer macro) that needs
967/// to be rewritten directly instead of passing an delegate attribute.
968///
969/// Especially in scenarios where attributes are not available to be passed for
970/// macro expansion.
971///
972/// The operation is:
973///
974/// ```text
975/// delegated-crate identifier
976///         |
977///         | replacement
978/// support-crate identifier
979/// ```
980///
981/// For example:
982///
983/// ```ignore
984/// // input syntax:
985/// outer_support::Trait
986/// outer_support::Type
987///
988/// // replacement:
989/// // outer_support -> inner_support
990///
991/// // resulting syntax:
992/// inner_support::Trait
993/// inner_support::Type
994/// ```
995///
996/// The implementation is expressed in terms of [`PathQualifierReplace`] so that the
997/// mechanism is independent of the particular `syn` syntax node being
998/// processed.
999///
1000/// This allows the same operation to be applied to different syntax
1001/// representations without making `SupportCrate` responsible for knowing
1002/// their concrete structure.
1003///
1004/// # Attribute delegation
1005///
1006/// [`Self::delegate_attribute`] is the deferred form of delegation. Instead
1007/// of immediately rewriting syntax, it places a delegate attribute onto the
1008/// syntax so that another macro expansion can occur first.
1009///
1010/// For example, if the syntax currently contains:
1011///
1012/// ```ignore
1013/// #[inner_support::outer_macro]
1014/// struct Marker;
1015/// ```
1016///
1017/// delegation can append:
1018///
1019/// ```ignore
1020/// #[inner_support::outer_macro]
1021/// #[inner_support::delegate_macro]
1022/// struct Marker;
1023/// ```
1024///
1025/// After the inner macro has expanded, the delegate macro receives the
1026/// resulting syntax and can perform the equivalent of:
1027///
1028/// ```text
1029/// outer_support -> inner_support
1030/// ```
1031///
1032/// This distinction is important because the two mechanisms solve different
1033/// expansion-order problems:
1034///
1035/// - direct delegation rewrites syntax that is already available;
1036/// - attribute delegation schedules the rewrite for syntax that will become
1037///   available only after an outer attribute macro expands.
1038///
1039/// # Using `delegate_inner` from a proc-macro attribute
1040///
1041/// [`Self::delegate_inner`] is intended to be called by the delegate attribute macro
1042/// provided by the proc-macro crate itself.
1043///
1044/// The delegate attribute is the bridge between the proc-macro expansion
1045/// mechanism and [`SupportCrate`]. The attribute receives the syntax produced
1046/// by the inner macro, parses that syntax into a `syn` representation, and
1047/// then asks the pipeline to perform the support-crate substitution.
1048///
1049/// For example, a proc-macro crate may expose a delegate attribute:
1050///
1051/// ```ignore
1052/// #[proc_macro_attribute]
1053/// pub fn delegate(attr: TokenStream, item: TokenStream) -> TokenStream {
1054///     let mut item: syn::Item = syn::parse_macro_input!(item);
1055///
1056///     Pipeline::delegate_inner(&mut item);
1057///
1058///     quote::quote!(#item).into()
1059/// }
1060/// ```
1061///
1062/// The important operation here is:
1063///
1064/// ```ignore
1065/// Pipeline::delegate_inner(&mut item);
1066/// ```
1067///
1068/// The delegate attribute does not need to know how delegate-crate identifiers
1069/// are represented or how they should be replaced towards support-crate.
1070/// That responsibility belongs to [`SupportCrate`].
1071///
1072/// For example, suppose the pipeline is configured with:
1073///
1074/// ```text
1075/// support crate  = inner_support
1076/// delegate crate = outer_support
1077/// ```
1078///
1079/// and the outer proc macro produces:
1080///
1081/// ```ignore
1082/// impl outer_support::Trait for Value {
1083///     fn method(&self) {}
1084/// }
1085/// ```
1086///
1087/// When the delegate attribute receives this generated item, it invokes:
1088///
1089/// ```ignore
1090/// Pipeline::delegate_inner(&mut item);
1091/// ```
1092///
1093/// which performs the configured identifier replacement:
1094///
1095/// ```text
1096/// outer_support
1097///       │
1098///       │ delegate_inner
1099///       │
1100/// inner_support
1101/// ```
1102///
1103/// resulting in:
1104///
1105/// ```ignore
1106/// impl inner_support::Trait for Value {
1107///     fn method(&self) {}
1108/// }
1109/// ```
1110///
1111/// This means the delegate attribute itself remains generic. It does not need
1112/// to contain a hard-coded replacement such as `outer_support -> inner_support`.
1113/// The pipeline supplies both identifiers through [`Self::support_crate`] and
1114/// [`Self::delegate_crate`].
1115///
1116/// The same mechanism can be used for any syntax representation implementing
1117/// [`PathQualifierReplace`]. For example, the delegate attribute may receive a
1118/// [`syn::Item`], [`syn::DeriveInput`], or another supported syntax node and
1119/// pass it directly to [`Self::delegate_inner`].
1120///
1121/// The complete responsibility is therefore divided into three parts:
1122///
1123/// ```text
1124/// proc-macro attribute
1125///        │
1126///        │ parses generated syntax
1127/// delegate_inner
1128///        │
1129///        │ obtains support_crate + delegate_crate
1130/// PathQualifierReplace
1131///        │
1132/// rewritten syntax
1133/// ```
1134///
1135/// This is the direct delegation path: the delegate attribute runs after the
1136/// inner macro has produced its syntax, and `delegate_inner` immediately
1137/// rewrites that already-available syntax.
1138pub trait SupportCrate {
1139    /// Returns the identifier of the normal support crate.
1140    ///
1141    /// Generated paths use this identifier to access the public support layer
1142    /// associated with the proc-macro pipeline.
1143    ///
1144    /// See [`SupportCrate`] documentation to understand the semantics.
1145    fn support_crate() -> syn::Ident;
1146
1147    /// Returns the crate identifier whose references should be replaced by the
1148    /// normal support crate when this pipeline is being delegated through another
1149    /// proc macro.
1150    ///
1151    /// `None` means that no crate substitution is required.
1152    ///
1153    /// See [`SupportCrate`] documentation to understand the semantics.
1154    fn delegate_crate() -> Option<syn::Ident> {
1155        None
1156    }
1157
1158    /// Returns the identifier of the attribute macro used to perform deferred
1159    /// support-crate delegation.
1160    ///
1161    /// This is used when an outer attribute macro generates syntax that cannot
1162    /// be rewritten by the inner macro directly.
1163    ///
1164    /// `None` means that no delegate attribute is available.
1165    ///
1166    /// See [`SupportCrate`] documentation to understand the semantics.
1167    fn delegate_macro() -> Option<syn::Ident> {
1168        None
1169    }
1170
1171    /// Replaces references to the delegated support crate with the normal
1172    /// support crate in an already available syntax representation.
1173    ///
1174    /// If no delegated crate is configured, the syntax is left unchanged.
1175    ///
1176    /// ```ignore
1177    /// replace_idents(
1178    ///     from: delegate_crate
1179    ///     to: support_crate
1180    /// )
1181    /// ```
1182    ///
1183    /// See [`SupportCrate`] documentation to understand the semantics.
1184    fn delegate_inner<T: PathQualifierReplace>(input: &mut T) {
1185        let Some(from) = Self::delegate_crate() else {
1186            return;
1187        };
1188
1189        let to = Self::support_crate();
1190
1191        input.replace_path_qualifiers(&mut ReplacePathQualifier { from, to });
1192    }
1193
1194    /// Appends the configured delegate attribute to a syntax representation.
1195    ///
1196    /// The delegate attribute is intentionally added rather than performing
1197    /// the replacement immediately. This allows an outer attribute macro to
1198    /// expand first, after which the delegate attribute can rewrite the
1199    /// delegate-crate references present in the generated syntax.
1200    ///
1201    /// If delegation has not been configured, this operation does nothing.
1202    ///
1203    /// See [`SupportCrate`] documentation to understand the semantics.
1204    fn delegate_attribute<T: PushAttribute>(input: &mut T) {
1205        let support_crate = Self::support_crate();
1206
1207        let Some(delegate_macro) = Self::delegate_macro() else {
1208            return;
1209        };
1210
1211        input.push_attribute(&mut AttributePush {
1212            attr: parse_quote!(#[#support_crate::#delegate_macro]),
1213        });
1214    }
1215
1216    /// Performs direct delegation for an item.
1217    ///
1218    /// Item delegation uses the same identifier-replacement mechanism as
1219    /// [`Self::delegate_inner`], because the item's syntax is available for
1220    /// direct inspection at the point where delegation is performed.
1221    fn delegate_item<T: PathQualifierReplace>(input: &mut T) {
1222        Self::delegate_inner(input);
1223    }
1224}
