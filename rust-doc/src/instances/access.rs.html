16//! Internal access traits used by the instances system to carry and recover
17//! compile-time metadata across generated instance implementations.
18//!
19//! This module defines the type-level access interfaces used by the
20//! [`crate`] re-exported proc-macros to communicate instance metadata through Rust's trait
21//! system. The access traits identify generated instance metadata by
22//! identifier hashes, contextual instance types, counter positions, and other
23//! compile-time parameters, allowing later generated code to recover metadata
24//! without revisiting the original instance definition.
25//!
26//! The access interfaces are implementation details of the instances macros
27//! and are intended to be used only by generated code produced by those macros.
28//! They are not intended for direct use by library users or by unrelated parts
29//! of the crate.
30//!
31//! The traits are public only where required so that generated implementations
32//! and generated access sites can cross the visibility boundaries imposed by
33//! Rust's type system. Their public visibility should therefore not be
34//! interpreted as a stable API guarantee.
35//!
36//! The signatures of these traits may change as the instances system evolves.
37//! Code outside the generated instances machinery should not depend on their
38//! names, generic parameter ordering, associated types, or other signature
39//! details.
40//!
41//! The module currently provides access to:
42//!
43//! - `CounterAccess` - Complete counter tuple for an identified instance.
44//! - `TerminalAccess` - Complete counter tuple for the terminal instance.
45//! - `BoundaryAccess` - Boundary counter tuple and final identifier hash.
46//! - `OnSetAccess` - Boundary counter tuple and initial identifier hash.
47//! - `IdentHashLenAccess` - Identifier length associated with a counter generic position.
48//!
49//! All metadata exposed through these interfaces is resolved at compile time
50//! through trait implementations generated by the instances macros.

63/// Resolves the counter tuple associated with an instance identified by one
64/// compile-time identifier hash.
65///
66/// `A` is the `u64` const value produced by an identifier-hashing expression,
67/// typically of the form `{ hash_ident(b"a-ident") }`. The hash identifies the
68/// instance impl implemented towards `SelfTy`.
69///
70/// The associated `Counter` type is the complete typenum counter tuple
71/// associated with the resolved instance. For a one-dimensional instance, the
72/// tuple contains the counter for dimension `0`.
73///
74/// For example, an implementation may associate:
75///
76/// ```ignore
77/// type Counter = (U2,);
78/// ```
79///
80/// The counter can then be projected from the resolved tuple through
81/// [`CountersGet<U0>`]:
82///
83/// ```ignore
84/// <Global as CounterAccess1<{ hash_ident(b"foo") }, InstanceType>>::Counter
85/// ```
86///
87/// resolves to `(U2,)`, whose `U0` element is `U2`.
88pub trait CounterAccess1<const A: u64, SelfTy: ?Sized> {
89    type Counter: CountersGet<U0>;
90}
91
92/// Resolves the counter tuple associated with an instance identified by a
93/// two-element sequence of compile-time identifier hashes.
94///
95/// `A` and `B` are `u64` const values produced by identifier-hashing
96/// expressions, typically of the form `{ hash_ident(b"ident") }`. Together
97/// they preserve the ordered identifier sequence used to identify the
98/// instance impl implemented towards `SelfTy`.
99///
100/// The associated `Counter` type is the complete typenum counter tuple
101/// associated with the resolved instance. The tuple contains one counter for
102/// each identifier dimension, with dimension `0` stored at `U0` and dimension
103/// `1` stored at `U1`.
104///
105/// For example, an implementation may associate:
106///
107/// ```ignore
108/// type Counter = (U0, U3);
109/// ```
110///
111/// The tuple can then be queried independently at each dimension:
112///
113/// ```ignore
114/// <Global as CounterAccess2<
115///     { hash_ident(b"foo") },
116///     { hash_ident(b"bar") },
117///     InstanceType,
118/// >>::Counter
119/// ```
120///
121/// provides `U0` through [`CountersGet<U0>`] and `U3` through
122/// [`CountersGet<U1>`].
123pub trait CounterAccess2<const A: u64, const B: u64, SelfTy: ?Sized> {
124    type Counter: CountersGet<U0> + CountersGet<U1>;
125}
126
127/// Resolves the counter tuple associated with an instance identified by a
128/// three-element sequence of compile-time identifier hashes.
129///
130/// `A`, `B`, and `C` are `u64` const values produced by identifier-hashing
131/// expressions, typically of the form `{ hash_ident(b"ident") }`. Their order
132/// is the order of the identifiers used to identify the instance impl
133/// implemented towards `SelfTy`.
134///
135/// The associated `Counter` type is the complete typenum counter tuple for
136/// the resolved instance. Each tuple position corresponds to the same
137/// dimension position in the identifier sequence.
138///
139/// For example:
140///
141/// ```text
142/// identifier hashes:
143///     A, B, C
144///
145/// counter tuple:
146///     (U0, U1, U2)
147/// ```
148///
149/// Each dimension is available through its corresponding [`CountersGet`]
150/// implementation, allowing later type-level operations to select the
151/// counter associated with an individual dimension.
152pub trait CounterAccess3<const A: u64, const B: u64, const C: u64, SelfTy: ?Sized> {
153    type Counter: CountersGet<U0> + CountersGet<U1> + CountersGet<U2>;
154}
155
156/// Resolves the counter tuple associated with an instance identified by a
157/// four-element sequence of compile-time identifier hashes.
158///
159/// `A`, `B`, `C`, and `D` are `u64` const values produced by identifier-hashing
160/// expressions, typically of the form `{ hash_ident(b"ident") }`. Their order
161/// is the order of the identifiers used to identify the instance impl
162/// implemented towards `SelfTy`.
163///
164/// The associated `Counter` type is the complete typenum counter tuple for
165/// the resolved instance. Each tuple position corresponds to the same
166/// dimension position in the identifier sequence.
167///
168/// For example:
169///
170/// ```text
171/// identifier hashes:
172///     A, B, C, D
173///
174/// counter tuple:
175///     (U0, U1, U2, U3)
176/// ```
177///
178/// Each dimension is available through its corresponding [`CountersGet`]
179/// implementation, allowing later type-level operations to select an
180/// individual counter from the resolved tuple.
181pub trait CounterAccess4<const A: u64, const B: u64, const C: u64, const D: u64, SelfTy: ?Sized> {
182    type Counter: CountersGet<U0> + CountersGet<U1> + CountersGet<U2> + CountersGet<U3>;
183}

189/// Resolves the counter tuple associated with the uniquely identified
190/// terminal instance impl implemented towards `SelfTy`.
191///
192/// Unlike other access traits, `TerminalAccess` requires no identifier hash
193/// sequence or counter-context generic arguments. The terminal instance is
194/// unique for the contextual instance type, so `SelfTy` alone is sufficient
195/// to select its generated access-trait implementation.
196///
197/// The associated `Terminal` type is the complete typenum counter tuple
198/// associated with the resolved terminal instance. Its elements represent the
199/// counters for the terminal instance's dimensions.
200///
201/// For example, a terminal access implementation may provide:
202///
203/// ```ignore
204/// impl TerminalAccess<InstanceType> for Global {
205///     type Terminal = (U2, U4);
206/// }
207/// ```
208///
209/// The counter tuple can then be recovered through the associated-type
210/// projection:
211///
212/// ```ignore
213/// <Global as TerminalAccess<InstanceType>>::Terminal
214/// ```
215///
216/// `TerminalAccess` therefore provides a unique access path to the terminal
217/// instance's complete counter context without requiring any identifier
218/// sequence or explicit counter dimensions.
219pub trait TerminalAccess<SelfTy: ?Sized> {
220    type Terminal: Marker;
221}

227/// Resolves the boundary counter tuple and the final identifier hash for the
228/// root identifier sequence.
229///
230/// `BoundaryAccess0` is used when there is no preceding identifier sequence.
231/// `SelfTy` alone therefore identifies the contextual instance whose
232/// identifiers form the boundary.
233///
234/// `FINAL` is the hash of the final identifier occurring at this boundary.
235///
236/// For example, if the identifiers at the root boundary are:
237///
238/// ```text
239/// a
240/// b
241/// c
242///
243/// FINAL = hash(c)
244/// ```
245///
246/// The associated `Boundary` type is the complete typenum counter tuple for
247/// the boundary and provides access to dimension `0` through
248/// [`CountersGet<U0>`].
249///
250/// ```ignore
251/// impl BoundaryAccess0<InstanceType> for Global {
252///     const FINAL: u64 = { hash_ident(b"c") };
253///     type Boundary = (U2,);
254/// }
255/// ```
256///
257/// The boundary information can then be recovered through:
258///
259/// ```ignore
260/// <Global as BoundaryAccess0<InstanceType>>::FINAL
261/// <Global as BoundaryAccess0<InstanceType>>::Boundary
262/// ```
263pub trait BoundaryAccess0<SelfTy: ?Sized> {
264    const FINAL: u64;
265    type Boundary: CountersGet<U0>;
266}
267
268/// Resolves the boundary counter tuple and the final identifier hash for a
269/// boundary identified by one preceding compile-time identifier hash.
270///
271/// `A` is the `u64` const value produced by an identifier-hashing expression,
272/// typically of the form `{ hash_ident(b"a-ident") }`. It identifies the
273/// preceding identifier sequence whose immediate children form this boundary.
274///
275/// `FINAL` is the hash of the final identifier occurring immediately after
276/// that preceding identifier sequence.
277///
278/// For example, given:
279///
280/// ```text
281/// a, b
282/// a, c
283/// a, k
284/// ```
285///
286/// the boundary identified by `a` has:
287///
288/// ```text
289/// INITIAL = hash(b)
290/// FINAL   = hash(k)
291/// ```
292///
293/// `BoundaryAccess1` exposes the final hash and the complete counter tuple for
294/// that boundary. The associated `Boundary` type provides access to dimensions
295/// `0` and `1` through [`CountersGet`].
296///
297/// ```ignore
298/// impl BoundaryAccess1<
299///     { hash_ident(b"a") },
300///     InstanceType,
301/// > for Global {
302///     const FINAL: u64 = { hash_ident(b"k") };
303///     type Boundary = (U0, U3);
304/// }
305/// ```
306pub trait BoundaryAccess1<const A: u64, SelfTy: ?Sized> {
307    const FINAL: u64;
308    type Boundary: CountersGet<U0> + CountersGet<U1>;
309}
310
311/// Resolves the boundary counter tuple and the final identifier hash for a
312/// boundary identified by a two-element sequence of compile-time identifier
313/// hashes.
314///
315/// `A` and `B` identify the preceding identifier sequence. The boundary
316/// consists of the identifiers that immediately extend that sequence.
317///
318/// For example, given:
319///
320/// ```text
321/// a, b, c
322/// a, b, k
323/// ```
324///
325/// the boundary identified by `a, b` has:
326///
327/// ```text
328/// INITIAL = hash(c)
329/// FINAL   = hash(k)
330/// ```
331///
332/// `FINAL` therefore identifies the last identifier occurring at the boundary,
333/// while the associated `Boundary` type contains the complete typenum counter
334/// tuple for the `a, b` prefix.
335///
336/// Each tuple position corresponds to the same dimension position in the
337/// preceding identifier sequence and can be accessed through the corresponding
338/// [`CountersGet`] implementation.
339pub trait BoundaryAccess2<const A: u64, const B: u64, SelfTy: ?Sized> {
340    const FINAL: u64;
341    type Boundary: CountersGet<U0> + CountersGet<U1> + CountersGet<U2>;
342}
343
344/// Resolves the boundary counter tuple and the final identifier hash for a
345/// boundary identified by a three-element sequence of compile-time identifier
346/// hashes.
347///
348/// `A`, `B`, and `C` identify the preceding identifier sequence. The boundary
349/// consists of the identifiers that immediately extend that sequence.
350///
351/// For example, given:
352///
353/// ```text
354/// a, b, c, d
355/// a, b, c, k
356/// ```
357///
358/// the boundary identified by `a, b, c` has:
359///
360/// ```text
361/// INITIAL = hash(d)
362/// FINAL   = hash(k)
363/// ```
364///
365/// `FINAL` therefore identifies the last identifier occurring at the boundary,
366/// while the associated `Boundary` type contains the complete typenum counter
367/// tuple for the `a, b, c` prefix.
368///
369/// Each tuple position corresponds to the same dimension position in the
370/// preceding identifier sequence and can be accessed through the corresponding
371/// [`CountersGet`] implementation.
372pub trait BoundaryAccess3<const A: u64, const B: u64, const C: u64, SelfTy: ?Sized> {
373    const FINAL: u64;
374    type Boundary: CountersGet<U0> + CountersGet<U1> + CountersGet<U2> + CountersGet<U3>;
375}

415    const INITIAL: u64;

417}

519/// Resolves the identifier length associated with a counter generic position
520/// within an instance identified by one compile-time identifier hash.
521///
522/// `A` is the `u64` const value produced by an identifier-hashing expression,
523/// typically of the form `{ hash_ident(b"a-ident") }`. The hash identifies the
524/// instance impl implemented towards `SelfTy`.
525///
526/// `GENERIC_IDX` selects the counter generic position whose corresponding
527/// identifier length is being accessed.
528///
529/// The associated `Length` type is the typenum representation of the
530/// identifier length at that generic position.
531///
532/// For example, given an instance identified by `A` with:
533///
534/// ```text
535/// identifier:       a
536/// generic index:    0
537/// identifier length: 3
538/// ```
539///
540/// the corresponding access may resolve as:
541///
542/// ```ignore
543/// <Global as IdentHashLenAccess1<
544///     { hash_ident(b"a") },
545///     0,
546///     InstanceType,
547/// >>::Length
548/// ```
549///
550/// and `Length` resolves to the typenum value representing `3`.
551///
552/// In most cases the length will be simply GENERIC_IDX + 1 in typenum.
553pub trait IdentHashLenAccess1<const A: u64, const GENERIC_IDX: usize, SelfTy: ?Sized> {
554    type Length: UnsignedTypeNum;
555}
556
557/// Resolves the identifier length associated with a counter generic position
558/// within an instance identified by a two-element sequence of compile-time
559/// identifier hashes.
560///
561/// `A` and `B` are `u64` const values produced by identifier-hashing
562/// expressions, typically of the form `{ hash_ident(b"ident") }`. Together
563/// they preserve the ordered identifier sequence used to identify the instance
564/// impl implemented towards `SelfTy`.
565///
566/// `GENERIC_IDX` selects the counter generic position whose corresponding
567/// identifier length is being accessed.
568///
569/// The associated `Length` type is the typenum representation of the
570/// identifier length at the selected generic position.
571///
572/// For example, an instance identified by:
573///
574/// ```text
575/// A, B
576/// ```
577///
578/// may have identifier lengths:
579///
580/// ```text
581/// generic index     identifier length
582///       0                    3
583///       1                    5
584/// ```
585///
586/// The corresponding access:
587///
588/// ```ignore
589/// <Global as IdentHashLenAccess2<
590///     { hash_ident(b"a") },
591///     { hash_ident(b"bbbbb") },
592///     1,
593///     InstanceType,
594/// >>::Length
595/// ```
596///
597/// resolves to the typenum value representing `5`.
598///
599/// In most cases the length will be simply GENERIC_IDX + 1 in typenum.
600pub trait IdentHashLenAccess2<const A: u64, const B: u64, const GENERIC_IDX: usize, SelfTy: ?Sized>
601{
602    type Length: UnsignedTypeNum;
603}
