16//! Recursive rewriting of instance-trait leaf-config access expressions.
17//!
18//! User-written instance impl accesses may appear directly as qualified
19//! paths or be nested inside higher-level Rust expressions such as
20//! function calls, tuples, references, `return` expressions, and `?`
21//! expressions.
22//!
23//! This module recursively traverses those expression forms until the
24//! underlying qualified instance-trait path is reached.
25//!
26//! Once located, the qualified path is rewritten to remove the counter
27//! const generics and supply the hidden typenum counter generic required
28//! to uniquely identify an instance implementation. That rewriting is
29//! performed by [`InstanceLeafAccessExprPath`], while the remaining
30//! transformations simply recurse through enclosing expression nodes
31//! until such a path is encountered.
32//!
33//! Only an instance trait implementing associated-type's node instance resolution
34//! is supported, allowing accesses or associated instance types to be rewritten uniformly.
35

61/// Recursively transforms instance-trait accesses within supported expression
62/// forms for leaf instance access.
63///
64/// The transformation traverses the supported expression wrappers, including
65/// function calls, tuples, references, `return` expressions, and `?`
66/// expressions, until the underlying qualified instance-trait path is reached.
67/// The qualified path is then rewritten by [`InstanceLeafAccessExprPath`] to
68/// resolve the associated item through the concrete leaf instance and its
69/// hidden typenum counter.
70///
71/// The supplied byte-string identifiers and counter indexes identify the
72/// requested instance position. Their lengths are required to match, and
73/// duplicate counter indexes are rejected before transformation begins.
74#[derive(Debug, Clone, Default)]
75pub(crate) struct InstanceLeafAccess;
76
77impl<'a> Transformation<AssocAccessorMut<'a>, (&BStringList, &IntList)> for InstanceLeafAccess {
78    fn validate_context(
79        &self,
80        context: &(&BStringList, &IntList),
81    ) -> Result<(), proc_macro2::TokenStream> {
82        context
83            .1
84            .duplicate_check(Some(ResolutionSpace::DuplicateCounterIndexes.into()))?;
85        Ok(())
86    }
87
88    fn raw_transform(
89        &self,
90        transform: &mut AssocAccessorMut<'a>,
91        context: &(&BStringList, &IntList),
92    ) -> Result<(), proc_macro2::TokenStream> {
93        match transform {
94            AssocAccessorMut::Path(expr_path) => InstanceLeafAccessExprPath::checked_transform(
95                &InstanceLeafAccessExprPath,
96                expr_path,
97                context,
98            )?,
99            AssocAccessorMut::Fn(expr_call) => {
100                InstanceDirectAccessExprCall::<Self>::checked_transform(
101                    &InstanceDirectAccessExprCall::<Self>::default(),
102                    expr_call,
103                    context,
104                )?
105            }
106            AssocAccessorMut::Tuple(expr_tuple) => {
107                InstanceDirectAccessExprTuple::<Self>::checked_transform(
108                    &InstanceDirectAccessExprTuple::<Self>::default(),
109                    expr_tuple,
110                    context,
111                )?
112            }
113            AssocAccessorMut::Ref(expr_reference) => {
114                InstanceDirectAccessExprRef::<Self>::checked_transform(
115                    &InstanceDirectAccessExprRef::<Self>::default(),
116                    expr_reference,
117                    context,
118                )?
119            }
120            AssocAccessorMut::Return(expr_return) => {
121                InstanceDirectAccessExprRetn::<Self>::checked_transform(
122                    &InstanceDirectAccessExprRetn::<Self>::default(),
123                    expr_return,
124                    context,
125                )?
126            }
127            AssocAccessorMut::Try(expr_try) => {
128                InstanceDirectAccessExprTry::<Self>::checked_transform(
129                    &InstanceDirectAccessExprTry::<Self>::default(),
130                    expr_try,
131                    context,
132                )?
133            }
134        }
135
136        Ok(())
137    }
138
139    fn validate_transform(
140        &self,
141        transform: &AssocAccessorMut<'a>,
142        context: Option<&(&BStringList, &IntList)>,
143    ) -> Result<(), proc_macro2::TokenStream> {
144        match transform {
145            AssocAccessorMut::Path(expr_path) => InstanceLeafAccessExprPath::validate_transform(
146                &InstanceLeafAccessExprPath,
147                expr_path,
148                context,
149            )?,
150            AssocAccessorMut::Fn(expr_call) => {
151                InstanceDirectAccessExprCall::<Self>::validate_transform(
152                    &InstanceDirectAccessExprCall::<Self>::default(),
153                    expr_call,
154                    context,
155                )?
156            }
157            AssocAccessorMut::Tuple(expr_tuple) => {
158                InstanceDirectAccessExprTuple::<Self>::validate_transform(
159                    &InstanceDirectAccessExprTuple::<Self>::default(),
160                    expr_tuple,
161                    context,
162                )?
163            }
164            AssocAccessorMut::Ref(expr_reference) => {
165                InstanceDirectAccessExprRef::<Self>::validate_transform(
166                    &InstanceDirectAccessExprRef::<Self>::default(),
167                    expr_reference,
168                    context,
169                )?
170            }
171            AssocAccessorMut::Return(expr_return) => {
172                InstanceDirectAccessExprRetn::<Self>::validate_transform(
173                    &InstanceDirectAccessExprRetn::<Self>::default(),
174                    expr_return,
175                    context,
176                )?
177            }
178            AssocAccessorMut::Try(expr_try) => {
179                InstanceDirectAccessExprTry::<Self>::validate_transform(
180                    &InstanceDirectAccessExprTry::<Self>::default(),
181                    expr_try,
182                    context,
183                )?
184            }
185        }
186        Ok(())
187    }
188}

194/// Rewrites qualified instance-trait path expressions to reference a
195/// specific instance implementation's associate item.
196///
197/// Instance traits internally replace their counter const generics with
198/// a hidden typenum counter type (both in [`traits`](crate::traits) and
199/// [`impls`](crate::impls)). Consequently, every qualified
200/// instance-trait path must be replaced with the corresponding typenum
201/// counter in order to uniquely identify the intended implementation.
202///
203/// This transformation derives that counter const generics via the supplied:
204///
205/// - counter identifiers,
206/// - counter-generic indexes,
207/// - and the implementing `Self` type,
208///
209/// removes them and inserts the recovered counter typenum type
210/// into the outermost qualified instance-trait path.
211///
212/// Qualified paths reference associated types:
213///
214/// ```ignore
215/// <<T as InstanceNode>::NodeAssoc as InstanceTrait<0,1>>::VALUE
216/// ```
217///
218/// Here the implementing `Self` type is **not** a concrete type but an
219/// associated type produced by another trait.
220///
221/// A trait declaring associated types that themselves implement instance
222/// traits is called a **instance node**.
223///
224/// For example:
225///
226/// ```ignore
227/// trait InstanceNode {
228///     type NodeAssoc: InstanceTrait<0,1>;
229/// }
230/// ```
231///
232/// Each such associated type is called an **subscriber node** (see [`crate::node`]).
233///
234/// Unlike a concrete implementing type, a subscriber node is not itself
235/// the canonical owner of its typenum counter tuple. Instead, every
236/// subscriber node exposes a hidden associated type providing access to
237/// the global counter lookup for each published instance node.
238///
239/// Conceptually:
240///
241/// ```ignore
242/// trait InstanceNode {
243///     type NodeAssoc: InstanceTrait<0,1>;
244///
245///     #[doc(hidden)]
246///     type __Global:
247///         CounterAccessN<HASH_0, HASH_1, Self::NodeAssoc>;
248/// }
249/// ```
250///
251/// The corresponding publisher implementation supplies both the concrete
252/// instance type and the global lookup type:
253///
254/// ```ignore
255/// impl InstanceNode for ConcreteNode {
256///     type NodeAssoc = InstanceType;
257///
258///     type __Global = Global;
259/// }
260/// ```
261///
262/// Consequently, subscriber resolution cannot directly construct:
263///
264/// ```ignore
265/// <Global as CounterAccessN<
266///     HASH_0,
267///     HASH_1,
268///     <T as InstanceNode>::NodeAssoc,
269/// >>::Counter
270/// ```
271///
272/// because an associated type is not the canonical source of the counter
273/// information. Instead, the transformation first recovers the hidden
274/// global lookup type exposed by the instance node:
275///
276/// ```ignore
277/// <T as InstanceNode>::__Global
278/// ```
279///
280/// and then performs the counter lookup through that recovered global
281/// type:
282///
283/// ```ignore
284/// <<T as InstanceNode>::__Global as CounterAccessN<
285///     HASH_0,
286///     HASH_1,
287///     <T as InstanceNode>::NodeAssoc,
288/// >>::Counter
289/// ```
290///
291/// Thus, publisher nodes (concrete node structs) provide the global lookup directly, whereas
292/// subscriber nodes first expose the appropriate global lookup through a
293/// hidden associated type before the counter tuple is recovered.
294///
295/// ## Result
296///
297/// This transformation rewrites the outermost qualified instance-trait path
298/// so that the hidden typenum counter argument is supplied automatically.
299///
300/// This preserves a single canonical mechanism for resolving instance
301/// implementations of associated instance types to participate uniformly
302/// in instance dispatch.
303#[derive(Debug, Clone, Default)]
304pub(crate) struct InstanceLeafAccessExprPath;
305
306impl<'a> Transformation<ExprPath, (&BStringList, &IntList)> for InstanceLeafAccessExprPath {
307    fn raw_transform(
308        &self,
309        transform: &mut ExprPath,
310        context: &(&BStringList, &IntList),
311    ) -> Result<(), proc_macro2::TokenStream> {
312        let expr = transform;
313        let idents = context.0;
314        let indexes = context.1;
315
316        if idents.bytes.len() != indexes.ints.len() {
317            return Err(ResolveErrors::IdentsAndIndexesNotSameLen {
318                indexes: indexes.clone(),
319                idents: idents.clone(),
320            }
321            .into());
322        }
323
324        let Some(q_base) = &mut expr.qself else {
325            return Err(ResolveErrors::ExprPathRequiresQSelf {
326                expr_path: expr.clone(),
327            }
328            .into());
329        };
330
331        let base_ty = &*q_base.ty;
332
333        let pos = q_base.position;
334        let trait_p = &mut expr.path.segments.iter_mut().take(pos);
335
336        let Some(trait_path) = trait_p.last() else {
337            return Err(ResolveErrors::InstanceTraitPathNotFound {
338                path: expr.path.clone(),
339            }
340            .into());
341        };
342
343        let Ok(assoc_path) = parse::<ExprPath>(base_ty.to_token_stream().into()) else {
344            return Err(ResolveErrors::QSelfRequiresExprPath {
345                qself: base_ty.clone(),
346            }
347            .into());
348        };
349
350        subscriber_call(&assoc_path, trait_path, indexes, idents)?;
351
352        Ok(())
353    }
354
355    fn validate_transform(
356        &self,
357        transform: &ExprPath,
358        context: Option<&(&BStringList, &IntList)>,
359    ) -> Result<(), proc_macro2::TokenStream> {
360        let Some(context) = context else {
361            return Ok(());
362        };
363
364        let expr = transform;
365        let idents = context.0;
366        let indexes = context.1;
367
368        if idents.bytes.len() != indexes.ints.len() {
369            return Err(ResolveBugs::IdentsAndIndexesNotSameLen {}.into());
370        }
371
372        let Some(q_base) = &expr.qself else {
373            return Err(ResolveBugs::ExprPathRequiresQSelf {}.into());
374        };
375
376        let base_ty = &q_base.ty;
377
378        let pos = q_base.position;
379        let trait_p = expr.path.segments.iter().take(pos);
380        let Some(trait_path) = trait_p.last() else {
381            return Err(ResolveBugs::InstanceTraitPathNotFound {}.into());
382        };
383
384        let Ok(assoc_path) = parse::<ExprPath>(base_ty.to_token_stream().into()) else {
385            return Err(ResolveBugs::QSelfRequiresExprPath {}.into());
386        };
387
388        validate_subscriber_call(&assoc_path, trait_path, idents, indexes)?;
389
390        Ok(())
391    }
392}
