13// ```````````````````` INSTANCE ASSOCIATE ENUMS (SUM TYPES) `````````````````````
14// ===============================================================================
15
16//! Provides sum-type boundaries for instance-dependent associated types.
17//!
18//! Every instance trait is extended with three hidden associated types:
19//! [`GlobalTerminalAssoc`], [`GlobalTerminalExact`], and [`SelfTerminalAssoc`]. Together,
20//! these identify the terminal instance associated with `Self`.
21//!
22//! Associated types marked with `#[sum]` are linked to the corresponding
23//! associated type of [`SelfTerminalAssoc`] through an associated-type equality
24//! constraint. The terminal-anchored instance is therefore required to expose
25//! the same associated types as the current instance.
26//!
27//! This makes the terminal instance a canonical type-level boundary rather
28//! than merely a conversion boundary. A `#[sum]` associated type on `Self` and
29//! the corresponding associated type on [`SelfTerminalAssoc`] are required to
30//! be the same type.
31//!
32//! The resulting sum type is therefore represented by a common terminal
33//! instance: different instance implementations may resolve through different
34//! instance anchors, while their `#[sum]` associated types are unified with
35//! the corresponding associated types of the terminal-anchored instance.
36//!
37//! In addition to establishing these equality relationships, this module
38//! removes the `#[sum]` marker after expansion so that it does not remain in
39//! the resulting trait definition.
40

69/// Identifies the global instance associated with the current instance.
70///
71/// For example, for:
72///
73/// ```ignore
74/// trait Example<I> { /* ... */ }
75/// ```
76///
77/// `GlobalTerminalAssoc` represents the associated type that provides the global
78/// instance for `Self`:
79///
80/// ```ignore
81/// type GlobalTerminalAssoc: TerminalAccess<Self>;
82/// ```
83///
84/// typically is `instances::Global` of the support crate of this
85/// proc macro crate.
86pub(crate) struct GlobalTerminalAssoc;
87
88/// Identifies the exact terminal counter of the global instance.
89///
90/// The terminal counter is obtained from [`GlobalTerminalAssoc`] and wrapped in
91/// `Exact`, giving the concrete counter used to construct the terminal
92/// instance.
93///
94/// For example:
95///
96/// ```ignore
97/// type GlobalTerminalExact:
98///     Exact<
99///         <Self::GlobalTerminalAssoc as TerminalAccess<Self>>::Terminal
100///     >;
101/// ```
102pub(crate) struct GlobalTerminalExact;
103
104/// Identifies the current trait instantiated with its terminal counter.
105///
106/// The instance counter corresponding to the first instance parameter is
107/// replaced with [`GlobalTerminalExact`].
108///
109/// For `#[sum]` associated types, the corresponding associated types of the
110/// terminal-anchored instance are additionally constrained to be equal to
111/// those of `Self`.
112///
113/// For example, given:
114///
115/// ```ignore
116/// trait Example<I> {
117///     #[sum]
118///     type Value;
119/// }
120/// ```
121///
122/// this represents:
123///
124/// ```ignore
125/// type SelfTerminalAssoc:
126///     Example<
127///         Self::GlobalTerminalExact,
128///         Value = Self::Value,
129///     >;
130/// ```
131///
132/// Thus, [`SelfTerminalAssoc`] represents the current trait instantiated at
133/// the terminal counter while preserving the `#[sum]` associated types through
134/// explicit associated-type equality constraints.
135///
136/// Usually, it is the `Self` implementation of the instance trait itself.
137pub(crate) struct SelfTerminalAssoc;

143/// Adds sum-type support to an instance trait by applying
144/// [`SumTypeExtension`] with its instance counter parameters.
145#[derive(Clone, Debug)]
146pub(crate) struct InstanceSumTypes;
147
148impl<'a> Transformation<ItemTrait, CounterParamsSlice<'a>> for InstanceSumTypes {
149    fn raw_transform(
150        &self,
151        transform: &mut ItemTrait,
152        context: &CounterParamsSlice<'a>,
153    ) -> Result<(), TokenStream> {
154        SumTypeExtension::checked_transform(&SumTypeExtension, transform, context)?;
155        Ok(())
156    }
157
158    fn validate_transform(
159        &self,
160        transform: &ItemTrait,
161        context: Option<&CounterParamsSlice<'a>>,
162    ) -> Result<(), TokenStream> {
163        SumTypeExtension::validate_transform(&SumTypeExtension, transform, context)?;
164        Ok(())
165    }
166}

168// ===============================================================================
169// `````````````````````````````` SUM TYPE EXTENSION `````````````````````````````
170// ===============================================================================

172/// Extends an instance trait with its terminal-anchored sum-type machinery.
173///
174/// For every instance trait, this extension adds three hidden associated types:
175///
176/// - [`GlobalTerminalAssoc`] - the global instance associated with `Self`.
177/// - [`GlobalTerminalExact`] - the exact terminal counter obtained from
178///   [`GlobalTerminalAssoc`].
179/// - [`SelfTerminalAssoc`] - the trait instance obtained by replacing the
180///   instance counter with [`GlobalTerminalExact`] and constraining each
181///   `#[sum]` associated type to be equal to the corresponding associated
182///   type of `Self`.
183///
184/// For example, given:
185///
186/// ```ignore
187/// trait Example<I> {
188///     #[sum]
189///     type Value;
190/// }
191/// ```
192///
193/// the extension conceptually adds:
194///
195/// ```ignore
196/// trait Example<I> {
197///     type GlobalTerminalAssoc: TerminalAccess<Self>;
198///
199///     type GlobalTerminalExact:
200///         Exact<<Self::GlobalTerminalAssoc as TerminalAccess<Self>>::Terminal>;
201///
202///     type SelfTerminalAssoc:
203///         Example<
204///             Self::GlobalTerminalExact,
205///             Value = Self::Value,
206///         >;
207/// }
208/// ```
209///
210/// Thus, [`SelfTerminalAssoc`] represents the same instance trait anchored to
211/// the terminal counter, with its `#[sum]` associated types explicitly unified
212/// with those of `Self`.
213///
214/// The resulting relationship is:
215///
216/// ```text
217/// Self
218///   |-- GlobalTerminalAssoc
219///          |-- GlobalTerminalExact
220///                    |
221///             SelfTerminalAssoc
222///                    | associated-type equality
223///              Self::#[sum] types
224/// ```
225///
226/// For every `#[sum]` associated type `Assoc`, the generated bound establishes:
227///
228/// ```text
229/// <Self::SelfTerminalAssoc as Trait<..., Self::GlobalTerminalExact, ...>>::Assoc
230///     ==
231/// Self::Assoc
232/// ```
233///
234/// Therefore, the instance-associated type and its terminal-instance
235/// counterpart are the same type rather than merely being convertible types.
236///
237/// This equality is established at the trait level, allowing the terminal
238/// associated type to be used wherever the corresponding instance associated
239/// type is expected without requiring an explicit `From` or `Into` conversion.
240///
241/// `SumTypeExtension` is responsible for establishing these type-level
242/// equality relationships; it does not itself define any runtime conversions.
243#[derive(Clone, Debug)]
244pub(crate) struct SumTypeExtension;
245
246impl<'a> Transformation<ItemTrait, CounterParamsSlice<'a>> for SumTypeExtension {
247    fn raw_transform(
248        &self,
249        transform: &mut ItemTrait,
250        context: &CounterParamsSlice<'a>,
251    ) -> Result<(), proc_macro2::TokenStream> {
252        let trait_ident = &transform.ident;
253        let crate_of = Instance::support_crate();
254
255        let Some(first) = context.first() else {
256            return Err(SumTypeBugs::CounterParamsAreEmpty {}.into());
257        };
258
259        let (_, ty_gen, _) = transform.generics.split_for_impl();
260
261        // GlobalTerminalAssoc
262
263        let global_assoc = gen_type_ident::<GlobalTerminalAssoc>();
264
265        transform.items.push(TraitItem::Type(TraitItemType {
266            attrs: proc_suite::internal_code(),
267            ident: global_assoc.clone(),
268            default: None,
269            type_token: Default::default(),
270            generics: Default::default(),
271            colon_token: Default::default(),
272            bounds: {
273                let mut bounds = Punctuated::new();
274
275                bounds.push(parse_quote!(
276                    #crate_of::TerminalAccess<Self>
277                ));
278
279                bounds
280            },
281            semi_token: Default::default(),
282        }));
283
284        // GlobalTerminalExact
285
286        let terminal_counter = gen_type_ident::<GlobalTerminalExact>();
287
288        transform.items.push(TraitItem::Type(TraitItemType {
289            attrs: proc_suite::internal_code(),
290            ident: terminal_counter.clone(),
291            default: None,
292            type_token: Default::default(),
293            generics: Default::default(),
294            colon_token: Default::default(),
295            bounds: {
296                let mut bounds = Punctuated::new();
297
298                bounds.push(parse_quote!(
299                    #crate_of::Exact<
300                        <Self::#global_assoc as #crate_of::TerminalAccess<Self>>::Terminal
301                    >
302                ));
303
304                bounds
305            },
306            semi_token: Default::default(),
307        }));
308
309        // SelfTerminalAssoc
310
311        let sum_inst = gen_type_ident::<SelfTerminalAssoc>();
312
313        let mut sum_inst_args: AngleBracketedGenericArguments = parse_quote!(#ty_gen);
314
315        let Some(arg) = sum_inst_args.args.iter_mut().nth(first.generic_index) else {
316            return Err(SumTypeBugs::TypeNumGenericNotFound {}.into());
317        };
318
319        *arg = GenericArgument::Type(parse_quote!(Self::#terminal_counter));
320
321        let mut collect = Vec::new();
322        for item in &transform.items {
323            let TraitItem::Type(ty) = item else {
324                continue;
325            };
326
327            if !ty.attrs.iter().any(|attr| attr.path().is_ident("sum")) {
328                continue;
329            }
330
331            let ty_ident = &ty.ident;
332
333            collect.push(ty_ident.clone());
334        }
335
336        let mut new_args = sum_inst_args.clone();
337        for ident in &collect {
338            new_args.args.push(parse_quote!(#ident = Self::#ident));
339        }
340
341        transform.items.push(TraitItem::Type(TraitItemType {
342            attrs: proc_suite::internal_code(),
343            ident: sum_inst.clone(),
344            default: None,
345            type_token: Default::default(),
346            generics: Default::default(),
347            colon_token: Default::default(),
348            bounds: {
349                let mut bounds = Punctuated::new();
350
351                bounds.push(parse_quote!(
352                    #trait_ident #new_args
353                ));
354
355                bounds
356            },
357            semi_token: Default::default(),
358        }));
359
360        for item in &mut transform.items {
361            let TraitItem::Type(ty) = item else {
362                continue;
363            };
364
365            let ty_ident = &ty.ident;
366
367            if !collect.iter().any(|try_ident| *try_ident == *ty_ident) {
368                continue;
369            }
370
371            let output: Type =
372                parse_quote!(<Self::#sum_inst as #trait_ident #sum_inst_args>::#ty_ident);
373            ty.bounds.push(parse_quote!(From<#output>));
374            ty.bounds.push(parse_quote!(Into<#output>));
375        }
376
377        Ok(())
378    }
379
380    fn validate_transform(
381        &self,
382        transform: &ItemTrait,
383        context: Option<&CounterParamsSlice<'a>>,
384    ) -> Result<(), proc_macro2::TokenStream> {
385        let crate_of = Instance::support_crate();
386
387        let global_assoc = gen_type_ident::<GlobalTerminalAssoc>();
388        let terminal_counter = gen_type_ident::<GlobalTerminalExact>();
389        let sum_inst = gen_type_ident::<SelfTerminalAssoc>();
390
391        let (_, ty_gen, _) = transform.generics.split_for_impl();
392
393        // GlobalTerminalAssoc
394
395        let global_assoc_bound: TraitBound = parse_quote!(
396            #crate_of::TerminalAccess<Self>
397        );
398
399        validate_trait_type! {
400            item: None,
401            towards: transform,
402            ident: global_assoc,
403            bounds: vec![&global_assoc_bound] => {
404                not_found: SumTypeBugs::GlobalAssocNotFound {},
405                wrong_ident: SumTypeBugs::GlobalAssocWrongIdent {},
406                has_generics: SumTypeBugs::GlobalAssocHasGenerics {},
407                bound_not_found: SumTypeBugs::GlobalAssocMissingBound {},
408            }
409        }?;
410
411        // GlobalTerminalExact
412
413        let terminal_counter_bound: TraitBound = parse_quote!(
414            #crate_of::Exact<
415                <Self::#global_assoc as #crate_of::TerminalAccess<Self>>::Terminal
416            >
417        );
418
419        validate_trait_type! {
420            item: None,
421            towards: transform,
422            ident: terminal_counter,
423            bounds: vec![&terminal_counter_bound] => {
424                not_found: SumTypeBugs::GlobalTerminalExactNotFound {},
425                wrong_ident: SumTypeBugs::GlobalTerminalExactWrongIdent {},
426                has_generics: SumTypeBugs::GlobalTerminalExactHasGenerics {},
427                bound_not_found: SumTypeBugs::GlobalTerminalExactMissingBound {},
428            }
429        }?;
430
431        // `SelfTerminalAssoc` must use the same trait as the current trait.
432        // The equality bindings (`Assoc = Self::Assoc`) are intentionally not
433        // included in `sum_inst_args`; they are validated separately below.
434
435        let trait_ident = &transform.ident;
436
437        validate_trait_type! {
438            item: None,
439            towards: transform,
440            ident: sum_inst,
441            bounds: Vec::<&TraitBound>::new() => {
442                not_found: SumTypeBugs::SelfTerminalNotFound {},
443                wrong_ident: SumTypeBugs::SelfTerminalWrongIdent {},
444                has_generics: SumTypeBugs::SelfTerminalHasGenerics {},
445                bound_not_found: SumTypeBugs::SelfTerminalMissingBound {},
446            }
447        }?;
448
449        // SelfTerminalAssoc
450        let Some(context) = context else {
451            return Ok(());
452        };
453
454        let Some(first) = context.first() else {
455            return Err(SumTypeBugs::CounterParamsAreEmpty {}.into());
456        };
457
458        let mut sum_inst_args: AngleBracketedGenericArguments = parse_quote!(#ty_gen);
459
460        let Some(arg) = sum_inst_args.args.iter_mut().nth(first.generic_index) else {
461            return Err(SumTypeBugs::TypeNumGenericNotFound {}.into());
462        };
463
464        *arg = GenericArgument::Type(parse_quote!(Self::#terminal_counter));
465
466        let mut collect = Vec::new();
467        for item in &transform.items {
468            let TraitItem::Type(ty) = item else {
469                continue;
470            };
471
472            if !ty.attrs.iter().any(|attr| attr.path().is_ident("sum")) {
473                continue;
474            }
475
476            let ty_ident = &ty.ident;
477
478            collect.push(ty_ident);
479        }
480
481        for ident in &collect {
482            sum_inst_args.args.push(parse_quote!(#ident = Self::#ident));
483        }
484
485        // Validate the trait bound of `SelfTerminalAssoc` manually.
486        //
487        // The expected bound is:
488        //
489        //     Trait<terminal arguments, Assoc = Self::Assoc, ...>
490        //
491        // Only the ordinary generic arguments are validated here. The associated-type
492        // equality bindings are generated from `collect` and are deliberately handled
493        // separately.
494
495        let sum_inst_item = transform.items.iter().find_map(|item| {
496            let TraitItem::Type(ty) = item else {
497                return None;
498            };
499
500            (ty.ident == sum_inst).then_some(ty)
501        });
502
503        let Some(sum_inst_item) = sum_inst_item else {
504            return Err(SumTypeBugs::SelfTerminalNotFound {}.into());
505        };
506
507        let Some(trait_bound) = sum_inst_item.bounds.iter().find_map(|bound| {
508            let syn::TypeParamBound::Trait(bound) = bound else {
509                return None;
510            };
511
512            (bound.path.segments.last()?.ident == *trait_ident).then_some(bound)
513        }) else {
514            return Err(SumTypeBugs::SelfTerminalMissingBound {}.into());
515        };
516
517        let Some(segment) = trait_bound.path.segments.last() else {
518            return Err(SumTypeBugs::SelfTerminalMissingBound {}.into());
519        };
520
521        let syn::PathArguments::AngleBracketed(args) = &segment.arguments else {
522            return Err(SumTypeBugs::SelfTerminalInvalidBound {}.into());
523        };
524
525        // `sum_inst_args` contains only the ordinary generic arguments.
526        // `args` contains those arguments followed by the associated-type equality
527        // bindings. Therefore, validate the ordinary arguments by comparing only
528        // the prefix corresponding to `sum_inst_args`.
529
530        let expected_args = &sum_inst_args.args;
531        let actual_args = &args.args;
532
533        if !(expected_args.len() <= actual_args.len()) {
534            return Err(SumTypeBugs::SelfTerminalInvalidBound {}.into());
535        }
536
537        for (expected, actual) in expected_args.iter().zip(actual_args.iter()) {
538            if expected != actual {
539                return Err(SumTypeBugs::SelfTerminalInvalidBound {}.into());
540            }
541        }
542
543        // The remaining arguments are the expected `#[sum]` associated-type
544        // equality bindings:
545        //
546        //     Assoc = Self::Assoc
547        //
548        // Their order is not significant. Each expected equality is therefore
549        // searched for independently among the actual associated-type arguments.
550        // The actual bound may contain additional non-equality arguments before,
551        // between, or after them.
552
553        let actual_equalities = actual_args.iter().filter_map(|arg| {
554            let GenericArgument::AssocType(assoc) = arg else {
555                return None;
556            };
557
558            Some(assoc)
559        });
560
561        for ident in &collect {
562            let expected_type: Type = parse_quote!(Self::#ident);
563
564            let Some(_) = actual_equalities
565                .clone()
566                .find(|assoc| assoc.ident == **ident && assoc.ty == expected_type)
567            else {
568                return Err(SumTypeBugs::SelfTerminalMissingBound {}.into());
569            };
570        }
571
572        Ok(())
573    }
574}

576// ===============================================================================
577// ``````````````````````````````` SUM ATTR REMOVAL ``````````````````````````````

580/// Removes the internal `#[sum]` marker from associated types.
581///
582/// And implies the associated type as a sum type via documentation regardless
583/// of the target usage.
584#[derive(Clone, Debug)]
585pub(crate) struct SumAttrRemoval;
586
587impl<'a> Transformation<ItemTrait> for SumAttrRemoval {
588    fn raw_transform(
589        &self,
590        transform: &mut ItemTrait,
591        _: &(),
592    ) -> Result<(), proc_macro2::TokenStream> {
593        for item in &mut transform.items {
594            let TraitItem::Type(ty) = item else {
595                let attrs = match item {
596                    TraitItem::Const(c) => &c.attrs,
597                    TraitItem::Fn(f) => &f.attrs,
598                    TraitItem::Macro(m) => &m.attrs,
599                    _ => {
600                        continue;
601                    }
602                };
603
604                if attrs.iter().any(|attr| attr.path().is_ident("sum")) {
605                    return Err(SumTypeErr::SumOnlyForAssocTypes { item: item.clone() }.into());
606                };
607
608                continue;
609            };
610
611            if ty.attrs.iter().any(|attr| attr.path().is_ident("sum")) {
612                // Remove #[sum], preserving all other attributes.
613                ty.attrs.retain(|attr| !attr.path().is_ident("sum"));
614            }
615        }
616        Ok(())
617    }
618
619    fn validate_transform(
620        &self,
621        transform: &ItemTrait,
622        _: Option<&()>,
623    ) -> Result<(), proc_macro2::TokenStream> {
624        for item in &transform.items {
625            let TraitItem::Type(ty) = item else {
626                continue;
627            };
628
629            for attr in &ty.attrs {
630                if attr.path().is_ident("sum") {
631                    return Err(SumTypeBugs::SumAttrNotRemoved {}.into());
632                }
633            }
634        }
635        Ok(())
636    }
