16//! Provides the transformation pipeline for instance access modules.
17//!
18//! The module transforms an instance access module containing inherent
19//! implementations for instance-node associated types. Each implementation
20//! is identified by a `#[node(<inst-args>)]` annotation that specifies the
21//! concrete instance arguments for the implemented instance node.
22//!
23//! The transformation resolves these instance arguments against the instance
24//! parameters declared by the access module and generates the corresponding
25//! instance access arguments used by the delegated access transformation.
26//!
27//! Each instance access implementation is represented in two forms: an
28//! internal `cfg(feature = "inst")` implementation for the actual 
29//! instance access transformation and a `cfg(not(feature = "inst"))` 
30//! implementation for rustdoc. The documentation representation includes 
31//! generated documentation describing the instance accessor and its arguments.
32//!
33//! Module-level documentation is generated separately by [`ModInstDoc`], while
34//! [`InstArguments`] renders the concrete instance arguments used by the
35//! documentation representation.
36//!
37//! The module also validates the transformed access implementations to ensure
38//! that internal and documentation targets are present in matching numbers,
39//! delegated transformation metadata is retained, and the generated
40//! implementations contain the required `cfg` attributes and documentation.
41

63/// Transforms an instance access module through the [`crate::inst`] macro
64/// pipeline.
65///
66/// The module contains inherent implementations annotated with
67/// `#[node(<inst-args>)]`. Each node annotation identifies the concrete
68/// instance arguments for the instance-node associated type implemented by
69/// that inherent implementation.
70///
71/// For each node implementation, this transformation:
72///
73/// 1. extracts and validates the instance arguments from `#[node(...)]`;
74/// 2. resolves those arguments against the instance parameters declared by
75///    the access module;
76/// 3. generates the corresponding instance access arguments;
77/// 4. attaches the explicit instance-access target attribute used by the
78///    delegated access transformation;
79/// 5. creates a `cfg(feature = "inst")` implementation for the actual 
80///    transformation;
81/// 6. creates a `cfg(not(feature = "inst"))` implementation for exposing 
82///    the documentation representation of the same instance access.
83/// 7. appends documentations for `cfg(not(feature = "inst"))` target for 
84///    each impl items for clarity.
85///
86/// The original `#[node(...)]` annotation is consumed during transformation
87/// and is not retained on either generated implementation.
88#[derive(Debug, Clone)]
89pub struct ModInst;
90
91impl Transformation<File, TokenStream> for ModInst {
92    fn raw_transform(
93        &self,
94        transform: &mut File,
95        context: &TokenStream,
96    ) -> Result<(), proc_macro2::TokenStream> {
97        if transform.items.len() > 1 {
98            return Err(ParseBug::TooManyItemsToTransform {}.into());
99        }
100        let Some(Item::Mod(mod_)) = transform
101            .items
102            .iter_mut()
103            .find(|item| matches!(item, Item::Mod(_)))
104        else {
105            return Err(ModBug::FileContainsNoMod {}.into());
106        };
107
108        let depth = InstAccess::checked_extract(context, &())?;
109
110        ModInst::checked_transform(&ModInst, mod_, &depth)?;
111        ModInstDoc::checked_transform(&ModInstDoc, mod_, &depth)?;
112
113        Ok(())
114    }
115
116    fn validate_transform(
117        &self,
118        transform: &File,
119        _: Option<&TokenStream>,
120    ) -> Result<(), proc_macro2::TokenStream> {
121        if transform.items.len() > 1 {
122            return Err(ModBug::TransformedFileInconsistent {}.into());
123        }
124        Ok(())
125    }
126}

132/// Transforms an instance access module into its documentation
133/// representation.
134///
135/// This transformation adds module-level documentation describing the
136/// instance access module, its instance-node inherent implementations, and the
137/// `inst_get!()` interface used to access them.
138///
139/// The documentation is retained on the transformed module and validated to
140/// ensure that the generated module documentation is present.
141#[derive(Debug, Clone)]
142pub struct ModInstDoc;
143
144impl Transformation<ItemMod, InstAccess> for ModInstDoc {
145    fn raw_transform(
146        &self,
147        transform: &mut ItemMod,
148        context: &InstAccess,
149    ) -> Result<(), proc_macro2::TokenStream> {
150        transform.attrs.prepend_docs(vec![
151            DocAttr::Heading {
152                level: 0,
153                content: vec![
154                    DocAttr::Raw("Instance Accessor Module ".to_string()),
155                    DocAttr::Ref(context.path.to_token_stream().to_string()),
156                    DocAttr::Raw(format!("(`{}`)", context.items.to_token_stream())),
157                ],
158            },
159            DocAttr::Raw(
160                "Source contains inherent implementations for instance-node \
161                 associated types, accessible through "
162                    .to_string(),
163            ),
164            DocAttr::Link {
165                title: "inst_get!()".to_string(),
166                url: "inst::inst_get".to_string(),
167            },
168            DocAttr::Raw(
169                " , since original methods of the trait cannot be accessed directly.".to_string(),
170            ),
171            DocAttr::LineBreak,
172            DocAttr::Raw("*See* ".to_string()),
173            DocAttr::Link {
174                title: "`inst::mod`".to_string(),
175                url: "inst::mod".to_string(),
176            },
177            DocAttr::Raw(" *for details.*".to_string()),
178            DocAttr::LineBreak,
179        ]);
180
181        Ok(())
182    }
183
184    fn validate_transform(
185        &self,
186        transform: &ItemMod,
187        _: Option<&InstAccess>,
188    ) -> Result<(), proc_macro2::TokenStream> {
189        if !transform.clone().has_docs() {
190            return Err(ModBug::ModuleDocsNotAppended {}.into());
191        }
192        Ok(())
193    }
194}

200/// Renders the instance arguments of a simple [`InstanceModel`] as tokens for
201/// documentation purposes.
202///
203/// Compile-time instance identifiers are emitted directly, while runtime
204/// instance parameters are rendered using the `dyn` argument form expected by
205/// instance accessors.
206///
207/// Complex instance models and inferred instance parameters are not valid for
208/// instance accessor arguments.
209struct InstArguments<'a>(&'a InstanceModel);
210
211impl ToTokens for InstArguments<'_> {
212    fn to_tokens(&self, tokens: &mut TokenStream) {
213        let InstanceModel::Simple(simple) = &self.0 else {
214            unreachable!("complex instance models are not possible for arguments");
215        };
216
217        for (i, param) in simple.params.iter().enumerate() {
218            if i != 0 {
219                tokens.extend(quote!(,));
220            }
221
222            match param {
223                crate::args::InstanceIdent::Compile(ident) => {
224                    ident.to_tokens(tokens);
225                }
226                crate::args::InstanceIdent::Runtime(dynamic_expr) => {
227                    let expr = &dynamic_expr.ident;
228                    quote!(dyn #expr).to_tokens(tokens);
229                }
230                crate::args::InstanceIdent::Infer(_) => {
231                    unreachable!("inferred instance parameters are not possible here");
232                }
233            }
234        }
235    }
236}

238/// Transforms each instance-node inherent implementation contained in the
239/// instance access module.
240///
241/// Implementations are selected by the presence of a `#[node(...)]`
242/// annotation. The annotation must contain a valid [`InstanceModel`], whose
243/// arguments are resolved against the instance-node declarations associated
244/// with the access module.
245///
246/// Each selected implementation is split into two forms: an internal
247/// `cfg(feature = "inst")` implementation carrying the actual instance-access 
248/// target, and a `cfg(not(feature = "inst"))` implementation carrying the 
249/// documentation target and generated accessor documentation for each impl items.
250///
251/// The transformed internal implementation replaces the original
252/// implementation, while the documentation implementation is appended to
253/// the module.
254/// 
255/// Additionally, to avoid documentation issues, every instance accessor function
256/// is replaced with a { todo!() } block during documentation builds, with a
257/// disclaimer to use --features inst for the actual implementation.
258impl Transformation<ItemMod, InstAccess> for ModInst {
259    fn raw_transform(
260        &self,
261        transform: &mut ItemMod,
262        context: &InstAccess,
263    ) -> Result<(), proc_macro2::TokenStream> {
264        let Some((_, items)) = &mut transform.content else {
265            return Err(ModError::RequiresInstAccessImpls {
266                module: transform.clone(),
267            }
268            .into());
269        };
270
271        let node = Inst::node();
272        let impl_indexes: Vec<usize> = items
273            .iter()
274            .enumerate()
275            .filter_map(|(index, item)| match item {
276                Item::Impl(impl_)
277                    if impl_.attrs.iter().any(|attr| attr.path().is_ident(&node)) =>
278                {
279                    Some(index)
280                }
281                _ => None,
282            })
283            .collect();
284
285        if impl_indexes.is_empty() {
286            return Err(ModError::RequiresInstAccessImpls {
287                module: transform.clone(),
288            }
289            .into());
290        }
291
292        for index in impl_indexes {
293            let mut impl_ = match &items[index] {
294                Item::Impl(impl_) => impl_.clone(),
295                _ => unreachable!(),
296            };
297
298            let mut node_attr = None;
299            let mut build = Vec::new();
300
301            for attr in &impl_.attrs {
302                if attr.path().is_ident(&node) {
303                    if node_attr.is_some() {
304                        return Err(ModError::DoubleNode { attr: attr.clone() }.into());
305                    }
306
307                    node_attr = Some(attr.clone());
308                } else {
309                    build.push(attr.clone());
310                }
311            }
312
313            impl_.attrs = build;
314
315            let Some(node_attr) = node_attr else {
316                return Err(ModBug::InconsistencyFindingNodeAttribute {}.into());
317            };
318
319            let tokens = match &node_attr.meta {
320                syn::Meta::List(meta_list) => &meta_list.tokens,
321
322                syn::Meta::Path(_) => {
323                    return Err(ModError::NodeRequiresArgs {
324                        attr: node_attr.clone(),
325                    }
326                    .into());
327                }
328
329                syn::Meta::NameValue(meta) => {
330                    return Err(ModError::NodeRequiresArgsNotValue { meta: meta.clone() }.into());
331                }
332            };
333
334            if tokens.is_empty() {
335                return Err(ModError::NodeRequiresArgs {
336                    attr: node_attr.clone(),
337                }
338                .into());
339            }
340
341            let inst = InstanceModel::checked_extract(tokens, &())?;
342
343            let support_crate = Inst::support_crate();
344
345            let direct_model = |impl_: &mut ItemImpl, not_doc: bool| -> Result<(), TokenStream> {
346                let Some(where_clause) = &mut impl_.generics.where_clause else {
347                    return Err(ModError::ExpectedWhereClauseForSelf {
348                        generics: impl_.generics.clone(),
349                    }
350                    .into());
351                };
352
353                let Some(bounds) = where_clause.predicates.iter_mut().find_map(|pred| {
354                    let WherePredicate::Type(ty) = pred else {
355                        return None;
356                    };
357
358                    if ty.bounded_ty != parse_quote!(Self) {
359                        return None;
360                    }
361
362                    Some(&mut ty.bounds)
363                }) else {
364                    return Err(ModError::ExpectedSelfInWhereClause {
365                        where_: where_clause.clone(),
366                        exp: context.path.clone(),
367                    }
368                    .into());
369                };
370
371                let indexes = append_placeholder_args(&mut *bounds, &context.path, &inst, not_doc)?;
372
373                let mut access = AccessArgsRef(inst.clone());
374
375                let args = access.try_access(&context.items, &indexes)?;
376
377                Inst::delegate_attribute(&mut impl_.attrs);
378
379                if not_doc {
380                    let access_target = Inst::access_target();
381                    impl_.attrs.insert(
382                        0,
383                        parse_quote!(
384                            #[#support_crate::#access_target(#args)]
385                        ),
386                    );
387                    impl_.attrs.insert(0, Inst::feature());
388                } else {
389                    let access_doc_target = Inst::access_doc_target();
390                    impl_.attrs.insert(
391                        0,
392                        parse_quote!(
393                            #[#support_crate::#access_doc_target(#args)]
394                        ),
395                    );
396                    impl_.attrs.insert(0, Inst::not_feature());
397                }
398
399                for item in &mut impl_.items {
400
401                    if !not_doc  {
402                        if let ImplItem::Fn(f) = item {
403                            f.block = parse_quote!(
404                                {
405                                    #[cfg(not(doc))]
406                                    {
407                                        compile_error!(
408                                            "`#[inst(...)]` macro requires the `inst` feature for build, \
409                                            recompile with `--features inst`"
410                                        );
411                                        todo!("expected build to be compiled with `--features inst`")
412                                    }
413
414                                    #[cfg(doc)] 
415                                    {
416                                        todo!("expected build to be compiled for documentation only");
417                                    }
418                                }
419                            );
420                        }
421                    }
422
423                    let attrs = match item {
424                        syn::ImplItem::Const(c) => &mut c.attrs,
425                        syn::ImplItem::Fn(f) => &mut f.attrs,
426                        syn::ImplItem::Type(t) => &mut t.attrs,
427                        _ => continue,
428                    };
429
430                    if not_doc {
431                        cfg_feature_disclaimer(attrs);
432                        attrs.insert(0,Inst::feature());
433                    } else {
434                        attrs.prepend_docs(vec![
435                            DocAttr::Heading {
436                                level: 0,
437                                content: vec![
438                                    DocAttr::Raw(
439                                        "Instance Accessor Item ".to_string(),
440                                    ),
441                                    DocAttr::Ref(
442                                        context
443                                            .path
444                                            .to_token_stream()
445                                            .to_string(),
446                                    ),
447                                    DocAttr::Raw(format!(
448                                        " `({})`",
449                                        InstArguments(&access.0)
450                                            .to_token_stream()
451                                            .to_string()
452                                    )),
453                                ],
454                            },
455                            DocAttr::Raw(
456                                "Instance accessor item with well-defined arguments. Dynamic arguments if exists (`dyn Param`) must be supplied through "
457                                    .to_string(),
458                            ),
459                            DocAttr::Link {
460                                title: "inst_get!()".to_string(),
461                                url: "inst::inst_get".to_string(),
462                            },
463                            DocAttr::Raw(
464                                " macro.".to_string(),
465                            ),
466                            DocAttr::LineBreak,
467                            DocAttr::Raw(
468                                "*See* ".to_string(),
469                            ),
470                            DocAttr::Link {
471                                title: "`inst::mod`".to_string(),
472                                url: "inst::mod".to_string(),
473                            },
474                            DocAttr::Raw(
475                                " *for details.*".to_string(),
476                            ),
477                            DocAttr::LineBreak,
478                        ]);
479                        attrs.insert(0,Inst::not_feature());
480
481                    }
482                }
483
484                impl_.attrs.retain(|attr| !attr.path().is_ident(&node));
485
486                Ok(())
487            };
488
489            let mut doc_impl = impl_.clone();
490            direct_model(&mut impl_, true)?;
491            direct_model(&mut doc_impl, false)?;
492            items[index] = Item::Impl(impl_);
493            items.push(Item::Impl(doc_impl));
494        }
495
496        Ok(())
497    }
498
499    fn validate_transform(
500        &self,
501        transform: &ItemMod,
502        _: Option<&InstAccess>,
503    ) -> Result<(), proc_macro2::TokenStream> {
504        let Some((_, items)) = &transform.content else {
505            return Err(ModBug::RequiresInstAccessImpls {}.into());
506        };
507
508        let support_crate = Inst::support_crate();
509        let access_target = Inst::access_target();
510        let access_doc_target = Inst::access_doc_target();
511        let Some(delegate_target) = Inst::delegate_macro() else {
512            return Err(ModBug::DelegateMacroNotInitialized {}.into())
513        };
514
515        let mut impl_items = Vec::new();
516        let mut impl_doc_items = Vec::new();
517        for item in items {
518            match item {
519                Item::Impl(impl_) => {
520                    if impl_.attrs.iter().any(|attr| {
521                        *attr.path()
522                            == parse_quote!(#support_crate::#access_target)
523                    }) {
524                        impl_items.push(impl_);
525                    }
526                    if impl_.attrs.iter().any(|attr| {
527                        *attr.path()
528                            == parse_quote!(#support_crate::#access_doc_target)
529                    }) {
530                        impl_doc_items.push(impl_);
531                    }
532                }
533                _ => {}
534            }
535        }
536
537        if impl_items.is_empty() || impl_doc_items.is_empty() {
538            return Err(ModBug::RequiresInstAccessImpls {}.into());
539        }
540
541        if impl_items.len() != impl_doc_items.len() {
542            return Err(ModBug::InconsistentTranformedImplTargetsLength {}.into());
543        }
544
545        let find = |doc: bool, attrs: &Vec<Attribute>| -> bool {
546            attrs.iter().any(|attr| {
547                if *attr == Inst::feature() {
548                    if doc {false} else {true}
549                } else if *attr == Inst::not_feature() {
550                    if doc {true} else {false}
551
552                } else {
553                    false
554                }
555            }) 
556        };
557
558        for impl_ in impl_items {
559            let Some((idx, _)) = impl_.attrs.iter().enumerate().find_map(|(i, attr)| {
560                if *attr.path() == parse_quote!(#support_crate::#access_target) {
561                    return Some((i, attr));
562                }
563                None
564            }) else {
565                unreachable!()
566            };
567
568            if !impl_.attrs.iter().enumerate().any(|(i, attr)| {
569                if i <= idx {
570                    return false;
571                };
572                *attr.path() == parse_quote!(#support_crate::#delegate_target)
573            }) {
574                return Err(ModBug::DelegateMacroNotFound {}.into());
575            }
576
577            for item in &impl_.items {
578                let attrs = match item {
579                    syn::ImplItem::Const(c) => &c.attrs,
580                    syn::ImplItem::Fn(f) => &f.attrs,
581                    syn::ImplItem::Type(t) => &t.attrs,
582                    _ => {
583                        continue;
584                    }
585                };
586
587                if !find(false, attrs) {
588                    return Err(ModBug::ItemNotDocTargetAttrNotFound {}.into());
589                }
590
591                if !attrs.clone().has_docs() {
592                    return Err(ModBug::ItemNotDocTargetHasNoDocs {}.into());
593                }
594            }
595        }
596
597        for impl_ in impl_doc_items {
598            let Some((idx, _)) = impl_.attrs.iter().enumerate().find_map(|(i, attr)| {
599                if *attr.path() == parse_quote!(#support_crate::#access_doc_target)
600                {
601                    return Some((i, attr));
602                }
603                None
604            }) else {
605                unreachable!()
606            };
607
608            if !impl_.attrs.iter().enumerate().any(|(i, attr)| {
609                if i <= idx {
610                    return false;
611                };
612                *attr.path() == parse_quote!(#support_crate::#delegate_target)
613            }) {
614                return Err(ModBug::DelegateMacroNotFound {}.into());
615            }
616
617            for item in &impl_.items {
618                let attrs = match item {
619                    syn::ImplItem::Const(c) => &c.attrs,
620                    syn::ImplItem::Fn(f) => &f.attrs,
621                    syn::ImplItem::Type(t) => &t.attrs,
622                    _ => {
623                        continue;
624                    }
625                };
626                if !find(true, attrs) {
627                    return Err(ModBug::ItemDocTargetAttrNotFound {}.into());
628                }
629
630                if !attrs.clone().has_docs() {
631                    return Err(ModBug::ItemDocTargetHasNoDocs {}.into());
632                }
633            }
634        }
635        Ok(())
636    }
