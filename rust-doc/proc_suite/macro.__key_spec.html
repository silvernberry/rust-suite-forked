macro_rules! __key_spec {
    (
        // Metadata for the marker key type
        $(#[$key_meta:meta])*
        $vis:vis $key_ty:ident => {
            key: $key:literal,
            allow_empty: $allow_empty:tt,

            sub_keys: {
                $($sub:ident => $sub_ty:tt),+ $(,)?
            },

            combinations: [
                $(
                    // Metadata for each enum variant
                    $(#[$variant_meta:meta])*
                    $variant:ident :
                    [ $($idx:expr),+ ]
                    =>
                    ( $($ty:ty),+ )
                    =>
                    ( $($field:ident),+ )
                ),+ $(,)?
            ]
        }
    ) => { ... };
    (
        @spec_enum
        vis = $vis:vis,
        allow_empty = true,
        key_meta = [$($key_meta:meta)*],
        key_ty = $key_ty:ident,
        key = $key:literal,
        variants = {
            $(
                $(#[$variant_meta:meta])*
                $variant:ident ( $( $ty:ty ),+ )
            ),+
        }
    ) => { ... };
    (
        @spec_enum
        vis = $vis:vis,
        allow_empty = false,
        key_meta = [$($key_meta:meta)*],
        key_ty = $key_ty:ident,
        key = $key:literal,
        variants = {
            $(
                $(#[$variant_meta:meta])*
                $variant:ident ( $( $ty:ty ),+ )
            ),+
        }
    ) => { ... };
    (
        @empty_variant_impl
        vis = $vis:vis,
        allow_empty = true,
        key_ty = $key_ty:ident,
    ) => { ... };
    (
        @empty_variant_impl
        vis = $vis:vis,
        allow_empty = false,
        key_ty = $key_ty:ident,
    ) => { ... };
}
Expand description

Internal implementation macro utilized by crate::key_schema!.

It is not intended to be invoked directly.

This macro expands a declarative key schema into:

  • a strongly-typed specification enum
  • its KeySpec implementation enforcing exact sub-key combinations

All external usage should go through crate::key_schema!, which provides a higher-level, user-facing syntax and generates the required supporting lookup macros.

This separation keeps the public schema definition ergonomic while allowing key_spec! to focus solely on deterministic decoding and diagnostics.
